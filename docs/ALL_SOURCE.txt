@tool
class_name CharSprite
extends Node2D
## PROVISIONAL ART. A tiny person drawn in code, origin at the feet.
## To be replaced by authored sprite sheets; keep set_motion() as the interface.

@export var coat := Color(0.35, 0.4, 0.45):
	set(v):
		coat = v
		queue_redraw()
@export var accent := Color(0.7, 0.45, 0.2):
	set(v):
		accent = v
		queue_redraw()
@export var skin := Color(0.78, 0.6, 0.48):
	set(v):
		skin = v
		queue_redraw()
@export var hood := false:
	set(v):
		hood = v
		queue_redraw()

var _dir := 0  # 0 down, 1 up, 2 left, 3 right
var _moving := false
var _t := 0.0
var _frame := 0


func set_motion(facing: Vector2, moving: bool) -> void:
	var d := _dir
	if facing != Vector2.ZERO:
		if absf(facing.x) > absf(facing.y):
			d = 2 if facing.x < 0.0 else 3
		else:
			d = 1 if facing.y < 0.0 else 0
	if d != _dir or moving != _moving:
		_dir = d
		_moving = moving
		if not moving:
			_frame = 0
			_t = 0.0
		queue_redraw()


func _process(delta: float) -> void:
	if Engine.is_editor_hint() or not _moving:
		return
	_t += delta
	var f := int(_t * 8.0) % 4
	if f != _frame:
		_frame = f
		queue_redraw()


func _draw() -> void:
	var bob := -1.0 if (_moving and _frame % 2 == 1) else 0.0
	var boots := Color(0.1, 0.1, 0.12)
	var dark := coat.darkened(0.22)
	var side := _dir >= 2
	draw_set_transform(Vector2.ZERO, 0.0, Vector2(1.0, 0.45))
	draw_circle(Vector2.ZERO, 6.0, Color(0, 0, 0, 0.3))
	draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)
	# legs
	var lift_a := 1.0 if (_moving and _frame == 1) else 0.0
	var lift_b := 1.0 if (_moving and _frame == 3) else 0.0
	if side:
		draw_rect(Rect2(-2 - lift_a * 2.0, -5, 2, 5 - lift_a), boots)
		draw_rect(Rect2(0 + lift_b * 2.0, -5, 2, 5 - lift_b), boots)
	else:
		draw_rect(Rect2(-3, -5, 2, 5 - lift_a), boots)
		draw_rect(Rect2(1, -5, 2, 5 - lift_b), boots)
	# coat
	var bw := 6.0 if side else 8.0
	draw_rect(Rect2(-bw * 0.5, -15 + bob, bw, 11), coat)
	draw_rect(Rect2(-bw * 0.5, -5 + bob, bw, 1), dark)
	draw_rect(Rect2(-bw * 0.5, -11 + bob, bw, 1), accent)
	if side:
		var swing := 0.0
		if _moving:
			swing = 1.0 if _frame < 2 else -1.0
		draw_rect(Rect2(-1 + swing, -14 + bob, 2, 6), dark)
	else:
		draw_rect(Rect2(-5, -14 + bob, 1, 6), dark)
		draw_rect(Rect2(4, -14 + bob, 1, 6), dark)
	# head
	var top := coat if hood else Color(0.14, 0.11, 0.1)
	draw_rect(Rect2(-3, -21 + bob, 6, 6), skin)
	draw_rect(Rect2(-4, -22 + bob, 8, 3), top)
	var eye := Color(0.08, 0.07, 0.08)
	match _dir:
		0:
			if hood:
				draw_rect(Rect2(-4, -20 + bob, 1, 5), top)
				draw_rect(Rect2(3, -20 + bob, 1, 5), top)
			draw_rect(Rect2(-2, -18 + bob, 1, 1), eye)
			draw_rect(Rect2(1, -18 + bob, 1, 1), eye)
		1:
			draw_rect(Rect2(-4, -22 + bob, 8, 7), top)
		2:
			draw_rect(Rect2(0, -22 + bob, 4, 7), top)
			draw_rect(Rect2(-3, -18 + bob, 1, 1), eye)
		3:
			draw_rect(Rect2(-4, -22 + bob, 4, 7), top)
			draw_rect(Rect2(2, -18 + bob, 1, 1), eye)

== art_provisional/fog.gdshader

shader_type canvas_item;
// PROVISIONAL. Drifting sea mist plus a soft vignette. One layer, no screen read.

uniform vec4 fog_color : source_color = vec4(0.6, 0.69, 0.75, 1.0);
uniform float fog_amount = 0.11;
uniform float vignette = 0.5;

float hash(vec2 p) {
	return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float noise(vec2 p) {
	vec2 i = floor(p);
	vec2 f = fract(p);
	f = f * f * (3.0 - 2.0 * f);
	return mix(mix(hash(i), hash(i + vec2(1.0, 0.0)), f.x),
			mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), f.x), f.y);
}

void fragment() {
	float n = noise(UV * vec2(3.0, 2.0) + vec2(TIME * 0.02, 0.0)) * 0.6
			+ noise(UV * vec2(7.0, 4.0) - vec2(TIME * 0.035, 0.0)) * 0.4;
	float fog = fog_amount * n * 1.6;
	float v = smoothstep(0.35, 0.85, distance(UV, vec2(0.5))) * vignette;
	vec3 col = mix(fog_color.rgb, vec3(0.01, 0.02, 0.04), v / (v + fog + 0.0001));
	COLOR = vec4(col, clamp(fog + v, 0.0, 0.9));
}

@tool
class_name ProvBlock
extends StaticBody2D
## PROVISIONAL ART. A solid piece of the world drawn in code, in 3/4 view.
## The node origin is the bottom-centre of the footprint (its front edge), so
## y-sorting works. `size` is the footprint (width x depth); `height` is how far
## the front face rises. Replace with authored art; keep the footprints.

@export var size := Vector2(48, 24):
	set(v):
		size = v
		queue_redraw()
@export var height := 20.0:
	set(v):
		height = v
		queue_redraw()
@export_enum("pod", "hut", "rock", "fence", "gate", "crate", "lamp", "tree", "pit", "adit", "sign", "none") var style := "crate":
	set(v):
		style = v
		queue_redraw()
@export var lit := false:
	set(v):
		lit = v
		queue_redraw()
@export var solid := true
@export var tint := Color(1, 1, 1):
	set(v):
		tint = v
		queue_redraw()

static var _light_tex: GradientTexture2D

var _shape: CollisionShape2D
var _solid_now := true


func _ready() -> void:
	if Engine.is_editor_hint():
		return
	collision_layer = 1
	collision_mask = 0
	if solid:
		var rect := RectangleShape2D.new()
		rect.size = size
		_shape = CollisionShape2D.new()
		_shape.shape = rect
		_shape.position = Vector2(0, -size.y * 0.5)
		_shape.disabled = not _solid_now
		add_child(_shape)
	if lit:
		_add_light()


func set_solid(v: bool) -> void:
	_solid_now = v
	if _shape != null:
		_shape.set_deferred("disabled", not v)


static func light_texture() -> GradientTexture2D:
	if _light_tex == null:
		var g := Gradient.new()
		g.set_color(0, Color(1, 1, 1, 1))
		g.set_color(1, Color(0, 0, 0, 0))
		_light_tex = GradientTexture2D.new()
		_light_tex.gradient = g
		_light_tex.width = 128
		_light_tex.height = 128
		_light_tex.fill = GradientTexture2D.FILL_RADIAL
		_light_tex.fill_from = Vector2(0.5, 0.5)
		_light_tex.fill_to = Vector2(1.0, 0.5)
	return _light_tex


func _add_light() -> void:
	var l := PointLight2D.new()
	l.texture = light_texture()
	l.color = Color(1.0, 0.6, 0.3)
	l.energy = 1.1
	if style == "lamp":
		l.position = Vector2(3, -height + 4)
		l.texture_scale = 1.9
	else:
		l.position = Vector2(size.x * 0.5 - 15, -height * 0.5 + 6)
		l.texture_scale = 1.2
	add_child(l)


func _draw() -> void:
	var w := size.x
	var d := size.y
	var h := height
	var l := -w * 0.5
	var rng := RandomNumberGenerator.new()
	rng.seed = hash(Vector2i(position))
	match style:
		"pod", "hut":
			var wall := (Color(0.4, 0.45, 0.48) if style == "pod" else Color(0.3, 0.36, 0.33)) * tint
			var roof := wall.lightened(0.14)
			draw_rect(Rect2(l, -d - h, w, d), roof)
			for i in range(1, int(w / 8.0)):
				draw_line(Vector2(l + i * 8, -d - h), Vector2(l + i * 8, -h), roof.darkened(0.14), 1.0)
			draw_rect(Rect2(l, -h, w, h), wall)
			draw_rect(Rect2(l, -h, w, 1), wall.darkened(0.35))
			draw_rect(Rect2(l, -2, w, 2), wall.darkened(0.4))
			draw_rect(Rect2(l, -h + 3, w, 1), Color(0.72, 0.4, 0.2, 0.75))
			draw_rect(Rect2(l + 6, -13, 8, 13), wall.darkened(0.5))
			draw_rect(Rect2(l + w - 23, -h + 6, 15, 8), wall.darkened(0.5))
			draw_rect(Rect2(l + w - 22, -h + 7, 13, 6), Color(1.0, 0.8, 0.45) if lit else Color(0.1, 0.14, 0.18))
			for i in 3:
				var x := l + rng.randf() * w
				draw_line(Vector2(x, -h + 4), Vector2(x, -h + 4 + rng.randf_range(4, 12)), Color(0.45, 0.27, 0.18, 0.5), 1.0)
		"rock":
			var face := Color(0.19, 0.22, 0.25) * tint
			draw_rect(Rect2(l, -d - h, w, d), Color(0.24, 0.28, 0.29) * tint)
			draw_rect(Rect2(l, -h, w, h), face)
			for i in int(w / 12.0):
				var p := Vector2(l + rng.randf() * w, -h + rng.randf() * h)
				draw_line(p, p + Vector2(rng.randf_range(6, 20), rng.randf_range(-2, 2)), face.lightened(0.1), 1.0)
				var q := Vector2(l + rng.randf() * w, -h + rng.randf() * h)
				draw_line(q, q + Vector2(rng.randf_range(-3, 3), rng.randf_range(5, 14)), face.darkened(0.3), 1.0)
			draw_rect(Rect2(l, -h - 1, w, 3), Color(0.15, 0.26, 0.2))
		"fence":
			var c := Color(0.37, 0.31, 0.27) * tint
			if w >= d:
				var n := maxi(1, int(w / 16.0))
				draw_rect(Rect2(l, -h + 2, w, h - 3), Color(c, 0.16))
				draw_line(Vector2(l, -h + 2), Vector2(l + w, -h + 2), c.lightened(0.15), 1.0)
				draw_line(Vector2(l, -h * 0.5), Vector2(l + w, -h * 0.5), c, 1.0)
				for i in n + 1:
					draw_rect(Rect2(l + i * (w / n) - 1, -h, 2, h), c)
			else:
				var n := maxi(1, int(d / 16.0))
				draw_line(Vector2(0, -h + 2), Vector2(0, -d - h + 2), c.lightened(0.15), 1.0)
				draw_line(Vector2(0, -h * 0.5), Vector2(0, -d - h * 0.5), c, 1.0)
				for i in n + 1:
					draw_rect(Rect2(-1, -i * (d / n) - h, 2, h), c)
		"gate":
			var c := Color(0.3, 0.33, 0.36)
			draw_rect(Rect2(l, -h, w, h), Color(c, 0.5))
			for i in range(1, int(w / 5.0)):
				draw_line(Vector2(l + i * 5, -h), Vector2(l + i * 5, 0), c, 1.0)
			draw_rect(Rect2(l, -h, w, 2), c.lightened(0.1))
			draw_rect(Rect2(l, -2, w, 2), c)
			draw_rect(Rect2(l, -h - 3, 3, h + 3), c.darkened(0.2))
			draw_rect(Rect2(l + w - 3, -h - 3, 3, h + 3), c.darkened(0.2))
			for i in int(w / 6.0):
				draw_rect(Rect2(l + i * 6, -h * 0.6, 6, 4), Color(0.8, 0.62, 0.15) if i % 2 == 0 else Color(0.08, 0.08, 0.09))
		"crate":
			var c := Color(0.55, 0.53, 0.47) * tint
			draw_rect(Rect2(l, -d - h, w, d), c.lightened(0.1))
			draw_rect(Rect2(l, -h, w, h), c.darkened(0.15))
			draw_line(Vector2(l, -h), Vector2(l + w, -h), c.darkened(0.4), 1.0)
			for i in range(1, 3):
				draw_line(Vector2(l, -h - d * i / 3.0), Vector2(l + w, -h - d * i / 3.0), c.darkened(0.2), 1.0)
		"lamp":
			var c := Color(0.2, 0.22, 0.24)
			draw_rect(Rect2(-1, -h, 2, h), c)
			draw_rect(Rect2(-1, -h, 6, 2), c)
			draw_rect(Rect2(2, -h + 2, 4, 2), Color(1.0, 0.82, 0.5) if lit else Color(0.12, 0.13, 0.15))
		"tree":
			var s := rng.randf_range(0.85, 1.2)
			var g := Color(0.08, 0.19, 0.16).lightened(rng.randf_range(0.0, 0.06)) * tint
			draw_set_transform(Vector2.ZERO, 0.0, Vector2(1.0, 0.4))
			draw_circle(Vector2.ZERO, 11.0 * s, Color(0, 0, 0, 0.3))
			draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)
			draw_rect(Rect2(-2, -12, 4, 12), Color(0.19, 0.14, 0.11))
			var tiers := 4
			var step := (h - 10.0) / tiers
			for i in tiers:
				var y0 := -9.0 - step * i
				var y1 := y0 - step * 1.7
				var hw := 15.0 * s * (1.0 - i * 0.2)
				draw_colored_polygon(PackedVector2Array([Vector2(-hw, y0), Vector2(hw, y0), Vector2(0, y1)]), g.lightened(0.05 * i))
				draw_colored_polygon(PackedVector2Array([Vector2(1, y0), Vector2(hw, y0), Vector2(1, y1)]), g.darkened(0.28))
		"pit":
			draw_rect(Rect2(l, -d, w, d), Color(0.035, 0.045, 0.055))
			draw_rect(Rect2(l, -d, w, 7), Color(0.14, 0.13, 0.12))
			draw_rect(Rect2(l, -d + 7, w, 2), Color(0.08, 0.08, 0.08))
			draw_rect(Rect2(l, -9, w, 5), Color(0.09, 0.15, 0.19))
		"adit":
			draw_rect(Rect2(l - 4, -h - 4, w + 8, h + 4), Color(0.25, 0.19, 0.14))
			draw_rect(Rect2(l, -h, w, h), Color(0.3, 0.33, 0.35))
			draw_line(Vector2(0, -h), Vector2(0, 0), Color(0.2, 0.22, 0.24), 1.0)
			draw_rect(Rect2(l, -h, w, h), Color(0.82, 0.84, 0.8), false, 1.0)
			for i in 4:
				draw_rect(Rect2(l + 4 + i * (w - 9) / 3.0, -h + 4, 1, 1), Color(0.15, 0.16, 0.18))
				draw_rect(Rect2(l + 4 + i * (w - 9) / 3.0, -5, 1, 1), Color(0.15, 0.16, 0.18))
		"sign":
			draw_rect(Rect2(l, -h, w, h - 5), Color(0.8, 0.78, 0.7))
			for i in 3:
				draw_line(Vector2(l + 1, -h + 2 + i * 2), Vector2(l + w - 1, -h + 2 + i * 2), Color(0.25, 0.22, 0.2), 1.0)
		_:
			pass

@tool
class_name ProvRail
extends Node2D
## PROVISIONAL ART. Narrow-gauge rail drawn along `points`. Also the path data
## that OreCart follows.

@export var points := PackedVector2Array():
	set(v):
		points = v
		queue_redraw()
@export var buffer_start := false
@export var buffer_end := false


func _draw() -> void:
	var tie := Color(0.2, 0.16, 0.13)
	var steel := Color(0.42, 0.43, 0.45)
	for i in points.size() - 1:
		var a := points[i]
		var b := points[i + 1]
		var dir := (b - a).normalized()
		var n := dir.orthogonal()
		var length := a.distance_to(b)
		var s := 0.0
		while s < length:
			var p := a + dir * s
			draw_line(p - n * 6.0, p + n * 6.0, tie, 2.0)
			s += 6.0
		draw_line(a - n * 4.0, b - n * 4.0, steel, 1.0)
		draw_line(a + n * 4.0, b + n * 4.0, steel, 1.0)
	if buffer_start and points.size() > 1:
		_buffer(points[0], (points[0] - points[1]).normalized())
	if buffer_end and points.size() > 1:
		var e := points.size() - 1
		_buffer(points[e], (points[e] - points[e - 1]).normalized())


func _buffer(at: Vector2, out: Vector2) -> void:
	var p := at + out * 12.0
	var n := out.orthogonal()
	draw_line(p - n * 7.0, p + n * 7.0, Color(0.5, 0.2, 0.15), 3.0)

class_name Rain
extends Control
## PROVISIONAL ART. Screen-space rain streaks.

@export var count := 110
@export var fall := Vector2(-70.0, 330.0)
@export var streak := 0.035

var _pos := PackedVector2Array()
var _speed := PackedFloat32Array()


func _ready() -> void:
	var rng := RandomNumberGenerator.new()
	rng.seed = 1
	for i in count:
		_pos.append(Vector2(rng.randf() * 700.0, rng.randf() * 300.0))
		_speed.append(rng.randf_range(0.75, 1.25))


func _process(delta: float) -> void:
	var s := size
	for i in _pos.size():
		var p := _pos[i] + fall * _speed[i] * delta
		if p.y > s.y:
			p.y -= s.y + 10.0
			p.x += 40.0
		if p.x < -10.0:
			p.x += s.x + 20.0
		elif p.x > s.x + 10.0:
			p.x -= s.x + 20.0
		_pos[i] = p
	queue_redraw()


func _draw() -> void:
	for i in _pos.size():
		draw_line(_pos[i], _pos[i] + fall * _speed[i] * streak, Color(0.75, 0.84, 0.92, 0.22), 1.0)

class_name Data
## JSON content loading. All narrative content lives under res://data.

static func load_json(path: String) -> Variant:
	if not FileAccess.file_exists(path):
		push_warning("Data: missing %s" % path)
		return null
	var parsed: Variant = JSON.parse_string(FileAccess.get_file_as_string(path))
	if parsed == null:
		push_error("Data: could not parse %s" % path)
	return parsed


static func load_dict(path: String) -> Dictionary:
	var v: Variant = load_json(path)
	if v is Dictionary:
		return v
	return {}

extends Node
## Data-driven dialogue runner. UI listens to the signals; nothing here draws.
##
## res://data/dialogue/<id>.json:
## {
##   "entry": [ {"if": cond, "goto": "node"}, {"goto": "first"} ],
##   "nodes": {
##     "first": {
##       "do":   [effects on entering the node],
##       "say":  [["Speaker", "line"], ["", "narration"]],
##       "then": [effects once the lines have been read],
##       "choices": [ {"text": "...", "if": cond, "do": [effects], "goto": "node"} ],
##       "goto": "node"            (used when no choice is available; default "end")
##     }
##   }
## }

signal started(id: String)
signal line_shown(speaker: String, text: String)
signal choices_shown(options: Array[String])
signal ended(id: String)

const ROOT := "res://data/dialogue/%s.json"

var active := false
var current_id := ""
var current_choices: Array[Dictionary] = []

var _data: Dictionary = {}
var _node: Dictionary = {}
var _lines: Array = []
var _line_i := 0
var _cache: Dictionary = {}
var _hops := 0


func start(id: String) -> bool:
	if active:
		return false
	if not _cache.has(id):
		_cache[id] = Data.load_dict(ROOT % id)
	_data = _cache[id]
	if _data.is_empty():
		push_warning("Dialogue: nothing to say for '%s'" % id)
		return false
	current_id = id
	active = true
	started.emit(id)
	var entry := "start"
	for e: Dictionary in _data.get("entry", []):
		if GameState.check(e.get("if")):
			entry = str(e.get("goto", "start"))
			break
	_hops = 0
	_enter(entry)
	return true


func advance() -> void:
	if not active or not current_choices.is_empty():
		return
	_line_i += 1
	if _line_i < _lines.size():
		_emit_line()
	else:
		_after_lines()


func choose(index: int) -> void:
	if not active or index < 0 or index >= current_choices.size():
		return
	var c := current_choices[index]
	current_choices = []
	GameState.apply(c.get("do"))
	_hops = 0
	_enter(str(c.get("goto", "end")))


func cancel() -> void:
	if active:
		_finish()


func choice_texts() -> Array[String]:
	var out: Array[String] = []
	for c in current_choices:
		out.append(str(c.get("text", "...")))
	return out


func _enter(node_name: String) -> void:
	var nodes: Dictionary = _data.get("nodes", {})
	_hops += 1
	if node_name == "end" or not nodes.has(node_name) or _hops > 32:
		if node_name != "end":
			push_warning("Dialogue '%s': bad node '%s'" % [current_id, node_name])
		_finish()
		return
	_node = nodes[node_name]
	GameState.apply(_node.get("do"))
	_lines = _node.get("say", [])
	_line_i = 0
	if _lines.is_empty():
		_after_lines()
	else:
		_emit_line()


func _emit_line() -> void:
	var l: Array = _lines[_line_i]
	line_shown.emit(str(l[0]), str(l[1]))


func _after_lines() -> void:
	GameState.apply(_node.get("then"))
	var opts: Array[Dictionary] = []
	for c: Dictionary in _node.get("choices", []):
		if GameState.check(c.get("if")):
			opts.append(c)
	if opts.is_empty():
		_enter(str(_node.get("goto", "end")))
	else:
		current_choices = opts
		choices_shown.emit(choice_texts())


func _finish() -> void:
	var id := current_id
	active = false
	current_id = ""
	current_choices = []
	_lines = []
	ended.emit(id)

== core/game_state.gd

extends Node
## Persistent world state: a flat dictionary of flags, plus the tiny condition/effect
## language shared by dialogue, interactables and puzzles.
##
## Conditions (Dictionary):
##   {"flag": "k"}                truthy flag
##   {"flag": "k", "eq": value}   flag equals value
##   {"clue": "id"}               journal holds the clue
##   {"not": cond} {"all": [..]} {"any": [..]}
## Effects (Array of Dictionary):
##   {"set": {"k": value}}  {"clue": "id"}  {"quest": "id"}
##   {"reveal": "quest.objective"}  {"objective": "quest.objective"}  {"complete": "quest"}
##   {"notify": "text"}

signal flag_changed(key: String, value: Variant)
signal notified(text: String)
## Asks the world (game/main.gd) to move the player to another map.
signal travel_requested(map_id: String, spawn: String)
## A named place was entered; the HUD shows it as a title card.
signal location_entered(text: String)

var flags: Dictionary = {}


func _ready() -> void:
	InputSetup.ensure()


func reset() -> void:
	flags = {}


func get_flag(key: String, default: Variant = false) -> Variant:
	return flags.get(key, default)


func set_flag(key: String, value: Variant) -> void:
	if flags.has(key) and _same(flags[key], value):
		return
	flags[key] = value
	flag_changed.emit(key, value)


func notify(text: String) -> void:
	notified.emit(text)


func check(cond: Variant) -> bool:
	if cond == null:
		return true
	if cond is Array:
		for c: Variant in cond:
			if not check(c):
				return false
		return true
	if not cond is Dictionary:
		return bool(cond)
	var d: Dictionary = cond
	if d.is_empty():
		return true
	if d.has("all"):
		return check(d["all"])
	if d.has("any"):
		for c: Variant in d["any"]:
			if check(c):
				return true
		return false
	if d.has("not"):
		return not check(d["not"])
	if d.has("clue"):
		return Journal.has_clue(str(d["clue"]))
	if d.has("flag"):
		var v: Variant = get_flag(str(d["flag"]))
		if d.has("eq"):
			return _same(v, d["eq"])
		return _truthy(v)
	push_warning("GameState.check: unknown condition %s" % [d])
	return true


func apply(effects: Variant) -> void:
	if not effects is Array:
		return
	for e: Variant in effects:
		if not e is Dictionary:
			continue
		var d: Dictionary = e
		if d.has("set"):
			var s: Dictionary = d["set"]
			for k: Variant in s:
				set_flag(str(k), s[k])
		if d.has("clue"):
			Journal.add_clue(str(d["clue"]))
		if d.has("quest"):
			Journal.start_quest(str(d["quest"]))
		if d.has("reveal"):
			Journal.reveal_objective(str(d["reveal"]))
		if d.has("objective"):
			Journal.complete_objective(str(d["objective"]))
		if d.has("complete"):
			Journal.complete_quest(str(d["complete"]))
		if d.has("notify"):
			notify(str(d["notify"]))


func to_dict() -> Dictionary:
	return flags.duplicate(true)


func from_dict(d: Dictionary) -> void:
	flags = d.duplicate(true)


static func _is_num(v: Variant) -> bool:
	return typeof(v) == TYPE_INT or typeof(v) == TYPE_FLOAT


static func _same(a: Variant, b: Variant) -> bool:
	if _is_num(a) and _is_num(b):
		return is_equal_approx(float(a), float(b))
	return typeof(a) == typeof(b) and a == b


static func _truthy(v: Variant) -> bool:
	match typeof(v):
		TYPE_NIL:
			return false
		TYPE_BOOL:
			return v
		TYPE_INT, TYPE_FLOAT:
			return v != 0
		TYPE_STRING:
			return v != ""
	return true

	_add_button(&"interact", JOY_BUTTON_A)
	_add_button(&"journal", JOY_BUTTON_Y)
	_add_button(&"menu", JOY_BUTTON_START)


static func _add(action: StringName, keys: Array, axis: int = -1, axis_value: float = 0.0) -> void:
	if InputMap.has_action(action):
		return
	InputMap.add_action(action, 0.25)
	for k: int in keys:
		var ev := InputEventKey.new()
		ev.physical_keycode = k as Key
		InputMap.action_add_event(action, ev)
	if axis >= 0:
		var jm := InputEventJoypadMotion.new()
		jm.axis = axis as JoyAxis
		jm.axis_value = axis_value
		InputMap.action_add_event(action, jm)


static func _add_button(action: StringName, button: JoyButton) -> void:
	var jb := InputEventJoypadButton.new()
	jb.button_index = button
	InputMap.action_add_event(action, jb)

extends Node
## Investigations (quests with objectives) and clues. Definitions are data
## (res://data/journal); this node only holds the player's progress.

signal updated(kind: String, text: String)

const HIDDEN := "hidden"
const ACTIVE := "active"
const DONE := "done"

var quest_defs: Dictionary = {}
var clue_defs: Dictionary = {}
## quest id -> {"objectives": {objective id: state}, "complete": bool}
var quests: Dictionary = {}
var clues: Array[String] = []


func _ready() -> void:
	quest_defs = Data.load_dict("res://data/journal/quests.json")
	clue_defs = Data.load_dict("res://data/journal/clues.json")


func reset() -> void:
	quests = {}
	clues.clear()


func start_quest(id: String) -> void:
	if quests.has(id) or not quest_defs.has(id):
		return
	var states := {}
	for o: Dictionary in quest_defs[id].get("objectives", []):
		states[str(o["id"])] = HIDDEN if o.get("hidden", false) else ACTIVE
	quests[id] = {"objectives": states, "complete": false}
	updated.emit("quest", quest_title(id))


func reveal_objective(path: String) -> void:
	var p := _split(path)
	if p.is_empty() or _state(p[0], p[1]) != HIDDEN:
		return
	quests[p[0]]["objectives"][p[1]] = ACTIVE
	updated.emit("objective_new", objective_text(p[0], p[1]))


func complete_objective(path: String) -> void:
	var p := _split(path)
	if p.is_empty() or _state(p[0], p[1]) in ["", DONE]:
		return
	quests[p[0]]["objectives"][p[1]] = DONE
	updated.emit("objective", objective_text(p[0], p[1]))


func complete_quest(id: String) -> void:
	if not quests.has(id) or quests[id]["complete"]:
		return
	quests[id]["complete"] = true
	updated.emit("complete", quest_title(id))


func add_clue(id: String) -> void:
	if clues.has(id) or not clue_defs.has(id):
		return
	clues.append(id)
	updated.emit("clue", str(clue_defs[id].get("title", id)))


func has_clue(id: String) -> bool:
	return clues.has(id)


func objective_state(quest: String, objective: String) -> String:
	return _state(quest, objective)


func is_complete(quest: String) -> bool:
	return quests.has(quest) and quests[quest]["complete"]


func quest_title(id: String) -> String:
	return str(quest_defs.get(id, {}).get("title", id))


func objective_text(quest: String, objective: String) -> String:
	for o: Dictionary in quest_defs.get(quest, {}).get("objectives", []):
		if str(o["id"]) == objective:
			return str(o.get("text", objective))
	return objective


func to_dict() -> Dictionary:
	return {"quests": quests.duplicate(true), "clues": clues.duplicate()}


func from_dict(d: Dictionary) -> void:
	quests = (d.get("quests", {}) as Dictionary).duplicate(true)
	clues.clear()
	for c: Variant in d.get("clues", []):
		clues.append(str(c))


func _split(path: String) -> PackedStringArray:
	var p := path.split(".")
	if p.size() != 2 or not quests.has(p[0]):
		return PackedStringArray()
	return p


func _state(quest: String, objective: String) -> String:
	if not quests.has(quest):
		return ""
	return str(quests[quest]["objectives"].get(objective, ""))

== core/save_manager.gd
==============================================================================
extends Node
## Save anywhere. One JSON file per slot under user://. "auto" is written whenever
## the app is backgrounded or closed and is resumed on boot, so a five-minute
## session never loses anything.
##
## The active world (game/main.gd) registers itself and provides
## capture_world() -> Dictionary and restore_world(Dictionary).

signal saved(slot: String)
signal loaded(slot: String)

const VERSION := 1

var world: Node = null
var autosave_enabled := true
var boot_from_autosave := true


func slot_path(slot: String) -> String:
	return "user://save_%s.json" % slot


func has_save(slot: String) -> bool:
	return FileAccess.file_exists(slot_path(slot))


func save_game(slot: String = "manual") -> bool:
	if world == null:
		return false
	var data := {
		"version": VERSION,
		"time": Time.get_datetime_string_from_system(),
		"flags": GameState.to_dict(),
		"journal": Journal.to_dict(),
		"world": world.call("capture_world"),
	}
	var path := slot_path(slot)
	var tmp := path + ".tmp"
	var f := FileAccess.open(tmp, FileAccess.WRITE)
	if f == null:
		push_error("SaveManager: cannot write %s" % tmp)
		return false
	f.store_string(JSON.stringify(data, "\t"))
	f.close()
	if FileAccess.file_exists(path):
		DirAccess.remove_absolute(path)
	if DirAccess.rename_absolute(tmp, path) != OK:
		push_error("SaveManager: cannot finalise %s" % path)
		return false
	saved.emit(slot)
	return true


func load_game(slot: String = "manual") -> bool:
	if world == null or not has_save(slot):
		return false
	var data: Dictionary = Data.load_dict(slot_path(slot))
	if data.is_empty() or int(data.get("version", 0)) != VERSION:
		push_warning("SaveManager: save '%s' is unreadable or from another version" % slot)
		return false
	GameState.from_dict(data.get("flags", {}))
	Journal.from_dict(data.get("journal", {}))
	world.call("restore_world", data.get("world", {}))
	loaded.emit(slot)
	return true


func delete_save(slot: String) -> void:
	if has_save(slot):
		DirAccess.remove_absolute(slot_path(slot))


func _notification(what: int) -> void:
	if what == NOTIFICATION_APPLICATION_PAUSED or what == NOTIFICATION_WM_CLOSE_REQUEST \
			or what == NOTIFICATION_WM_GO_BACK_REQUEST:
		if autosave_enabled and world != null:
			save_game("auto")

	_enter_map(id)
	player.global_position = map.spawn_position(spawn)
	if player.walk_mask != null:
		player.global_position = player.walk_mask.nearest_walkable(player.global_position)
	player.camera.reset_smoothing()


func capture_world() -> Dictionary:
	return {
		"map": map_id,
		"x": player.global_position.x,
		"y": player.global_position.y,
		"fx": player.facing.x,
		"fy": player.facing.y,
	}


func restore_world(d: Dictionary) -> void:
	var id := str(d.get("map", START_MAP))
	if not MAPS.has(id):
		id = START_MAP
	_enter_map(id)
	player.global_position = Vector2(float(d.get("x", 0.0)), float(d.get("y", 0.0)))
	player.facing = Vector2(float(d.get("fx", 0.0)), float(d.get("fy", 1.0)))
	player.camera.reset_smoothing()

		world.remove_child(map)

	world.add_child(map)
	player = PLAYER_SCENE.instantiate() as Player
	map.entities.add_child(player)
	player.setup(ui.touch, map.bounds)
	player.walk_mask = map.get_node_or_null("WalkMask") as WalkMask
	map.player = player
	ui.bind_player(player)
	ui.hud.show_location(map.display_name)

	if MAPS.has(id):
		load_map.call_deferred(id, spawn)

		player.try_interact()

class_name Player
extends CharacterBody2D

signal target_changed(target: Interactable)

@export var walk_speed := 64.0

var touch: TouchControls
var facing := Vector2.DOWN
var target: Interactable
## Optional painted walkable area (maps without collision bodies).
var walk_mask: WalkMask

@onready var sprite: CharSprite = $Sprite
@onready var reach: Area2D = $Reach
@onready var camera: Camera2D = $Camera


func setup(touch_controls: TouchControls, bounds: Rect2) -> void:
	touch = touch_controls
	camera.limit_left = int(bounds.position.x)
	camera.limit_top = int(bounds.position.y)
	camera.limit_right = int(bounds.end.x)
	camera.limit_bottom = int(bounds.end.y)
	camera.reset_smoothing()


func _physics_process(_delta: float) -> void:
	var dir := Vector2.ZERO
	if not Dialogue.active:
		dir = Input.get_vector(&"move_left", &"move_right", &"move_up", &"move_down")
		if dir == Vector2.ZERO and touch != null:
			dir = touch.vector
	var moving := dir.length() > 0.1
	velocity = _allowed(dir.limit_length(1.0) * walk_speed, _delta) if moving else Vector2.ZERO
	move_and_slide()
	if moving:
		facing = dir.normalized()
	sprite.set_motion(facing, moving)
	_update_target()


func _allowed(v: Vector2, delta: float) -> Vector2:
	if walk_mask == null or not _clear(global_position):
		return v
	if _clear(global_position + v * delta):
		return v
	var vx := Vector2(v.x, 0.0)
	if v.x != 0.0 and _clear(global_position + vx * delta):
		return vx
	var vy := Vector2(0.0, v.y)
	if v.y != 0.0 and _clear(global_position + vy * delta):
		return vy
	return Vector2.ZERO


func _clear(p: Vector2) -> bool:
	return walk_mask.is_walkable(p) and walk_mask.is_walkable(p + Vector2(3, 0)) \
			and walk_mask.is_walkable(p + Vector2(-3, 0)) and walk_mask.is_walkable(p + Vector2(0, -3))


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed(&"interact") and try_interact():
		get_viewport().set_input_as_handled()


func try_interact() -> bool:
	if Dialogue.active or target == null or not target.is_available():
		return false
	target.interact(self)
	return true


func _update_target() -> void:
	var best: Interactable = null
	var best_d := INF
	for a in reach.get_overlapping_areas():
		var it := a as Interactable
		if it == null or not it.is_available():
			continue
		var d := global_position.distance_squared_to(it.global_position)
		if d < best_d:
			best_d = d
			best = it
	if best != target:
		target = best
		target_changed.emit(target)

class_name OreCart
extends StaticBody2D
## A cart on a rail spur. Pushed from its start it rolls to the junction and takes
## whichever branch the points are set to: branch A ends at a buffer (haul it back
## and try again), branch B ends in the washout, where the cart wedges and becomes
## a bridge. State lives in GameState so it survives save/load.
## PROVISIONAL ART (drawn in code).

@export var state_key := "anjabi.cart"
@export var switch_key := "anjabi.spur_switch"
@export var rail_main: NodePath
@export var rail_a: NodePath
@export var rail_b: NodePath
## ProvBlock pit segment whose collision is removed once the cart bridges it.
@export var gap: NodePath
@export var bridge_position := Vector2.ZERO
@export var roll_speed := 70.0
@export var dialogue_buffer := ""
@export var dialogue_bridge := ""
@export var on_bridged: Array[Dictionary] = []

var moving := false

var _interactable: Interactable
var _shape: CollisionShape2D


func _ready() -> void:
	collision_layer = 1
	collision_mask = 0
	var rect := RectangleShape2D.new()
	rect.size = Vector2(18, 10)
	_shape = CollisionShape2D.new()
	_shape.shape = rect
	_shape.position = Vector2(0, -5)
	add_child(_shape)
	_interactable = Interactable.new()
	_interactable.label = "Ore cart"
	_interactable.radius = 16.0
	_interactable.position = Vector2(0, -5)
	add_child(_interactable)
	_interactable.interacted.connect(_on_interacted)
	_apply_state()


func state() -> String:
	return str(GameState.get_flag(state_key, "start"))


func _points(path: NodePath) -> PackedVector2Array:
	var rail := get_node(path) as ProvRail
	var out := PackedVector2Array()
	for p in rail.points:
		out.append(rail.to_global(p))
	return out


func _apply_state() -> void:
	var s := state()
	var bridged := s == "trench"
	match s:
		"buffer":
			var a := _points(rail_a)
			global_position = a[a.size() - 1]
		"trench":
			global_position = bridge_position
		_:
			global_position = _points(rail_main)[0]
	_shape.set_deferred("disabled", bridged)
	z_index = -4 if bridged else 0
	_interactable.enabled = not bridged
	_interactable.prompt = "Haul back" if s == "buffer" else "Push"
	var pit := get_node_or_null(gap) as ProvBlock
	if pit != null:
		pit.set_solid(not bridged)
	queue_redraw()


func _on_interacted(_by: Node2D) -> void:
	if moving:
		return
	var route := PackedVector2Array()
	var next := "start"
	var main := _points(rail_main)
	if state() == "start":
		var to_trench := str(GameState.get_flag(switch_key, "a")) == "b"
		route.append_array(main.slice(1))
		route.append_array(_points(rail_b if to_trench else rail_a).slice(1))
		if to_trench:
			route.append(bridge_position)
		next = "trench" if to_trench else "buffer"
	elif state() == "buffer":
		var a := _points(rail_a)
		a.reverse()
		route.append_array(a.slice(1))
		main.reverse()
		route.append_array(main.slice(1))
	else:
		return
	_roll(route, next)


func _roll(route: PackedVector2Array, next: String) -> void:
	moving = true
	_interactable.enabled = false
	var tween := create_tween()
	var from := global_position
	for i in route.size():
		var t := from.distance_to(route[i]) / roll_speed
		var step := tween.tween_property(self, "global_position", route[i], maxf(t, 0.05))
		if i == 0:
			step.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN)
		from = route[i]
	await tween.finished
	moving = false
	GameState.set_flag(state_key, next)
	_apply_state()
	if next == "trench":
		GameState.apply(on_bridged)
		if dialogue_bridge != "":
			Dialogue.start(dialogue_bridge)
	elif next == "buffer" and dialogue_buffer != "":
		Dialogue.start(dialogue_buffer)


func _draw() -> void:
	var rust := Color(0.5, 0.29, 0.18)
	if state() == "trench" and not moving:
		# Wedged in the washout, bed up: a short steel deck.
		draw_rect(Rect2(-13, -42, 26, 44), Color(0.23, 0.2, 0.19))
		draw_rect(Rect2(-12, -41, 24, 42), rust.darkened(0.2))
		for i in 6:
			draw_line(Vector2(-12, -38 + i * 7), Vector2(12, -38 + i * 7), rust.darkened(0.45), 1.0)
		return
	draw_rect(Rect2(-8, -3, 4, 3), Color(0.1, 0.1, 0.11))
	draw_rect(Rect2(4, -3, 4, 3), Color(0.1, 0.1, 0.11))
	draw_colored_polygon(PackedVector2Array([
		Vector2(-11, -15), Vector2(11, -15), Vector2(8, -3), Vector2(-8, -3)]), rust)
	draw_rect(Rect2(-11, -17, 22, 3), rust.lightened(0.12))
	draw_rect(Rect2(-9, -16, 18, 1), Color(0.08, 0.08, 0.09))
	draw_line(Vector2(-3, -13), Vector2(-4, -5), rust.darkened(0.3), 1.0)
	draw_line(Vector2(5, -13), Vector2(4, -6), rust.darkened(0.3), 1.0)

class_name WalkMask
extends Node2D
## Walkable area stored as a BitMap, for maps laid out from a painted plan
## rather than from collision bodies. `pixel_size` is world pixels per mask pixel.

@export var mask: BitMap
@export var pixel_size := 2.0


func is_walkable(world: Vector2) -> bool:
	if mask == null:
		return true
	var p := Vector2i(((world - global_position) / pixel_size).floor())
	var s := mask.get_size()
	if p.x < 0 or p.y < 0 or p.x >= s.x or p.y >= s.y:
		return false
	return mask.get_bitv(p)


func nearest_walkable(world: Vector2, max_radius: float = 240.0) -> Vector2:
	if is_walkable(world):
		return world
	var r := pixel_size
	while r <= max_radius:
		for i in 16:
			var p := world + Vector2.from_angle(TAU * i / 16.0) * r
			if is_walkable(p):
				return p
		r += pixel_size
	return world

@tool
extends Node2D
## PROVISIONAL ART. Ground painting for Anjabi / Collapsed Mine (location 04), drawn once.
## Layout follows Greybox Study 01: stairs down from the Estate (north-west), a walled
## yard on a cliff ledge, a path east toward the airpad, and a drop to the south.

func _draw() -> void:
	var rng := RandomNumberGenerator.new()
	rng.seed = 7
	var earth := Color(0.17, 0.22, 0.2)
	var gravel := Color(0.27, 0.29, 0.3)
	var mud := Color(0.29, 0.26, 0.23)
	var rock := Color(0.17, 0.2, 0.22)
	draw_rect(Rect2(0, 0, 960, 704), earth)
	draw_rect(Rect2(900, 0, 60, 704), Color(0.08, 0.14, 0.13))
	draw_rect(Rect2(0, 0, 112, 704), rock)
	# inner yard and the corridor behind the washout
	draw_rect(Rect2(200, 106, 700, 144), gravel)
	draw_rect(Rect2(742, 250, 158, 50), gravel)
	draw_rect(Rect2(456, 106, 48, 144), mud.darkened(0.12))
	# stairs down from the Estate
	var stone := Color(0.4, 0.4, 0.38)
	draw_rect(Rect2(122, 0, 76, 252), stone)
	for i in 38:
		draw_line(Vector2(122, 16 + i * 6), Vector2(198, 16 + i * 6), stone.darkened(0.3), 1.0)
	draw_rect(Rect2(122, 0, 2, 252), stone.darkened(0.45))
	draw_rect(Rect2(196, 0, 2, 252), stone.darkened(0.45))
	# landing, path to the gate, haul road, wash-house apron, path east to the airpad
	draw_rect(Rect2(116, 250, 90, 44), mud)
	draw_colored_polygon(PackedVector2Array([Vector2(200, 258), Vector2(454, 254), Vector2(454, 292), Vector2(200, 294)]), mud)
	draw_colored_polygon(PackedVector2Array([Vector2(450, 690), Vector2(510, 690), Vector2(506, 250), Vector2(454, 250)]), mud)
	draw_colored_polygon(PackedVector2Array([Vector2(150, 596), Vector2(452, 584), Vector2(452, 620), Vector2(150, 628)]), mud.darkened(0.08))
	draw_colored_polygon(PackedVector2Array([Vector2(508, 626), Vector2(900, 630), Vector2(900, 662), Vector2(508, 658)]), mud.darkened(0.05))
	draw_rect(Rect2(762, 340, 126, 46), Color(0.21, 0.25, 0.23))
	for x: float in [466.0, 494.0]:
		draw_line(Vector2(x, 690), Vector2(x, 250), mud.darkened(0.22), 2.0)
	# the drop to the south: rock lip, then mist
	draw_rect(Rect2(0, 690, 960, 14), rock.darkened(0.25))
	for i in 5:
		draw_rect(Rect2(0, 694 + i * 2, 960, 2), Color(0.45, 0.52, 0.58, 0.1 + i * 0.07))
	# puddles
	for p: Vector3 in [Vector3(472, 470, 9), Vector3(492, 360, 7), Vector3(470, 620, 11), Vector3(640, 644, 8),
			Vector3(330, 606, 7), Vector3(300, 170, 10), Vector3(640, 200, 8), Vector3(820, 270, 6),
			Vector3(480, 180, 9), Vector3(160, 272, 8)]:
		draw_set_transform(Vector2(p.x, p.y), 0.0, Vector2(1.0, 0.4))
		draw_circle(Vector2.ZERO, p.z, Color(0.33, 0.42, 0.5, 0.85))
		draw_circle(Vector2(-p.z * 0.3, -p.z * 0.2), p.z * 0.35, Color(0.5, 0.6, 0.68, 0.5))
	draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)
	# gravel flecks, weeds, rock cracks
	for i in 900:
		var p := Vector2(rng.randf_range(114, 898), rng.randf_range(108, 688))
		if p.x < 200 and p.y < 252:
			continue
		var on_road := p.x > 448 and p.x < 512
		if p.y < 250 or (p.x > 742 and p.y < 300):
			draw_rect(Rect2(p, Vector2(1, 1)), gravel.lightened(rng.randf_range(-0.15, 0.2)))
		elif not on_road:
			var c := earth.lightened(rng.randf_range(0.02, 0.14))
			draw_line(p, p + Vector2(rng.randf_range(-1, 1), -rng.randf_range(2, 4)), c, 1.0)
	for i in 60:
		var p := Vector2(rng.randf_range(4, 104), rng.randf_range(4, 700))
		draw_line(p, p + Vector2(rng.randf_range(-6, 6), rng.randf_range(6, 18)), rock.lightened(0.08), 1.0)

extends GameMap
## Anjabi, whole settlement, greybox. Ground art and walkable area both come from
## docs/reference/anjabi_greybox_01.png (see tools/build_town_mask.gd).
## Announces the district the player is standing in.

## Image-pixel rectangles on the greybox; first match wins.
const DISTRICTS := [
	["THE STEPS", Rect2(820, 370, 125, 75)],
	["TRAM STATION", Rect2(940, 440, 180, 210)],
	["SHOPS", Rect2(890, 220, 170, 160)],
	["MAIN TAVERN", Rect2(670, 190, 215, 175)],
	["TION TOWER", Rect2(425, 300, 180, 175)],
	["GUARD TOWER", Rect2(930, 40, 210, 165)],
	["WORKER PODS", Rect2(440, 40, 325, 205)],
	["ESTATE GARDENS", Rect2(95, 320, 215, 100)],
	["ESTATE", Rect2(120, 420, 290, 165)],
	["COLLAPSED MINE", Rect2(60, 630, 250, 180)],
	["PRIVATE AIRPAD", Rect2(320, 735, 200, 130)],
	["BEACH", Rect2(500, 560, 430, 175)],
	["DOCKS AND BOARDWALK", Rect2(1060, 360, 320, 215)],
	["RAINFOREST APPROACH", Rect2(1255, 220, 165, 150)],
	["FOREST", Rect2(40, 40, 330, 215)],
]

var _district := ""
var _acc := 0.0

@onready var _mask: WalkMask = $WalkMask


func _process(delta: float) -> void:
	_acc += delta
	if _acc < 0.25 or player == null:
		return
	_acc = 0.0
	var p := (player.global_position - _mask.global_position) / _mask.pixel_size
	var found := ""
	for d: Array in DISTRICTS:
		if (d[1] as Rect2).has_point(p):
			found = d[0]
			break
	if found != _district:
		_district = found
		if found != "":
			GameState.location_entered.emit("ANJABI  ·  " + found)

## Plays the open dialogue: picks the first choice containing any of `prefer`
## (each used once), otherwise the last choice, until it ends.
func _talk(prefer: Array) -> void:
	var want := prefer.duplicate()
	for i in 200:
		if not Dialogue.active:
			break
		var opts := Dialogue.choice_texts()
		if opts.is_empty():
			Dialogue.advance()
		else:
			var pick := opts.size() - 1
			for w: String in want:
				var found := false
				for j in opts.size():
					if opts[j].contains(w):
						pick = j
						want.erase(w)
						found = true
						break
				if found:
					break
			Dialogue.choose(pick)
		await get_tree().process_frame
	await _frames(2)

extends SceneTree
## Builds the walkable mask for the Anjabi town greybox from the layout image itself.
##   Godot --headless --path . -s tools/build_town_mask.gd
## Walkways, beach, docks (beige) and vegetation (green) are walkable; cliffs,
## buildings and water are not. Light-grey slabs that are meant to be walked on
## cannot be told apart from buildings by colour, so they are listed in SLABS.
## Re-run after replacing maps/anjabi/town_greybox.png.

const SRC := "res://maps/anjabi/town_greybox.png"
const OUT := "res://maps/anjabi/town_walk.res"
const DEBUG := "res://docs/reference/town_walkable_debug.png"
const FRAME := Rect2i(34, 34, 1380, 902)
const GREEN_WALKABLE := true
const CLOSE_RADIUS := 3
## Image-pixel polygons forced walkable (mine yard, airpad).
const SLABS := [
	[Vector2(132, 655), Vector2(215, 642), Vector2(295, 668), Vector2(300, 715), Vector2(240, 768), Vector2(150, 755)],
	[Vector2(327, 790), Vector2(357, 750), Vector2(508, 750), Vector2(508, 826), Vector2(425, 850), Vector2(327, 800)],
]
## Landmarks (image pixels) checked for reachability from the first one.
const LANDMARKS := {
	"Steps": Vector2(880, 415), "Forest": Vector2(230, 110), "Garden": Vector2(200, 345),
	"Estate": Vector2(240, 578), "Mine": Vector2(212, 700), "Airpad": Vector2(450, 800),
	"Beach": Vector2(650, 640), "Tion Tower": Vector2(515, 465), "Tavern": Vector2(760, 350),
	"Pods": Vector2(580, 70), "Guard Tower": Vector2(945, 130), "Docks": Vector2(1170, 460),
	"Shop": Vector2(930, 385), "Tram": Vector2(970, 465), "Rainforest": Vector2(1380, 325),
}

var w := 0
var h := 0


func _init() -> void:
	var img := Image.load_from_file(ProjectSettings.globalize_path(SRC))
	img.convert(Image.FORMAT_RGB8)
	w = img.get_width()
	h = img.get_height()
	var px := img.get_data()
	var m := PackedByteArray()
	m.resize(w * h)
	for y in range(FRAME.position.y, FRAME.end.y):
		for x in range(FRAME.position.x, FRAME.end.x):
			var i := (y * w + x) * 3
			var r := px[i]
			var g := px[i + 1]
			var b := px[i + 2]
			var beige := r > 205 and g > 185 and b < 228 and r - b > 22 and r >= g
			var green := g > r + 12 and g > b + 25
			if beige or (GREEN_WALKABLE and green):
				m[y * w + x] = 1
	for poly: Array in SLABS:
		var pts := PackedVector2Array(poly)
		var box := Rect2(pts[0], Vector2.ZERO)
		for p in pts:
			box = box.expand(p)
		for y in range(int(box.position.y), int(box.end.y) + 1):
			for x in range(int(box.position.x), int(box.end.x) + 1):
				if Geometry2D.is_point_in_polygon(Vector2(x, y), pts):
					m[y * w + x] = 1
	# close small gaps: label text, stair treads, outlines
	m = _spread(m, CLOSE_RADIUS, 1)
	m = _spread(m, CLOSE_RADIUS, 0)
	for y in h:
		for x in w:
			if not FRAME.has_point(Vector2i(x, y)):
				m[y * w + x] = 0

	var bm := BitMap.new()
	bm.create(Vector2i(w, h))
	var dbg := Image.load_from_file(ProjectSettings.globalize_path(SRC))
	dbg.convert(Image.FORMAT_RGB8)
	var reach := _flood(m, _nearest(m, LANDMARKS["Steps"]))
	for y in h:
		for x in w:
			var i := y * w + x
			if m[i] == 1:
				bm.set_bit(x, y, true)
				dbg.set_pixel(x, y, Color(0.2, 0.9, 0.4) if reach[i] == 1 else Color(0.95, 0.2, 0.7))
			else:
				dbg.set_pixel(x, y, dbg.get_pixel(x, y).darkened(0.55))
	ResourceSaver.save(bm, OUT)
	dbg.save_png(ProjectSettings.globalize_path(DEBUG))
	for k: String in LANDMARKS:
		var p := _nearest(m, LANDMARKS[k])
		var ok := p.x >= 0 and reach[p.y * w + p.x] == 1
		print("%-12s %s  (snapped %.0f px)" % [k, "reachable" if ok else "NOT REACHABLE", Vector2(p).distance_to(LANDMARKS[k])])
	quit()


## Separable box dilation of cells equal to `value`.
func _spread(src: PackedByteArray, r: int, value: int) -> PackedByteArray:
	var tmp := src.duplicate()
	for y in h:
		var run := 0
		for x in range(-r, w):
			var add := x + r
			if add < w and src[y * w + add] == value:
				run += 1
			var drop := x - r - 1
			if drop >= 0 and src[y * w + drop] == value:
				run -= 1
			if x >= 0 and run > 0:
				tmp[y * w + x] = value
	var out := tmp.duplicate()
	for x in w:
		var run := 0
		for y in range(-r, h):
			var add := y + r
			if add < h and tmp[add * w + x] == value:
				run += 1
			var drop := y - r - 1
			if drop >= 0 and tmp[drop * w + x] == value:
				run -= 1
			if y >= 0 and run > 0:
				out[y * w + x] = value
	return out


func _nearest(m: PackedByteArray, at: Vector2) -> Vector2i:
	var c := Vector2i(at)
	for r in range(0, 60):
		for dy in range(-r, r + 1):
			for dx in range(-r, r + 1):
				if maxi(absi(dx), absi(dy)) != r:
					continue
				var p := c + Vector2i(dx, dy)
				if p.x >= 0 and p.y >= 0 and p.x < w and p.y < h and m[p.y * w + p.x] == 1:
					return p
	return Vector2i(-1, -1)


func _flood(m: PackedByteArray, start: Vector2i) -> PackedByteArray:
	var seen := PackedByteArray()
	seen.resize(w * h)
	if start.x < 0:
		return seen
	var stack := PackedInt32Array([start.y * w + start.x])
	seen[stack[0]] = 1
	while not stack.is_empty():
		var i := stack[stack.size() - 1]
		stack.remove_at(stack.size() - 1)
		var x := i % w
		for n: int in [i - 1, i + 1, i - w, i + w]:
			if n < 0 or n >= w * h or seen[n] == 1 or m[n] == 0:
				continue
			if (n == i - 1 and x == 0) or (n == i + 1 and x == w - 1):
				continue
			seen[n] = 1
			stack.append(n)
	return seen

#!/usr/bin/env python3
"""Concatenate every text source file into docs/ALL_SOURCE.txt for reviewers and
tools that cannot browse the repo. Run from the project root:
    python3 tools/bundle_source.py
"""
import os

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OUT = os.path.join(ROOT, "docs", "ALL_SOURCE.txt")
EXTS = (".gd", ".gdshader", ".json", ".tscn", ".godot", ".py", ".sh")
SKIP_DIRS = {".git", ".godot", "export", "docs"}

paths = []
for base, dirs, files in os.walk(ROOT):
    dirs[:] = sorted(d for d in dirs if d not in SKIP_DIRS)
    for f in sorted(files):
        if f.endswith(EXTS):
            paths.append(os.path.relpath(os.path.join(base, f), ROOT))

with open(OUT, "w") as out:
    out.write("COMETO - all source files. Index:\n")
    for p in paths:
        out.write("  %s\n" % p)
    for p in paths:
        out.write("\n\n" + "=" * 78 + "\n== %s\n" % p + "=" * 78 + "\n")
        with open(os.path.join(ROOT, p)) as f:
            out.write(f.read())
print("wrote %s (%d files)" % (os.path.relpath(OUT, ROOT), len(paths)))

	_more.text = "▾"
	_more.horizontal_alignment = HORIZONTAL_ALIGNMENT_RIGHT
	_more.add_theme_color_override("font_color", UiTheme.AMBER)
	_more.add_theme_font_size_override("font_size", 8)
	box.add_child(_more)

	Dialogue.started.connect(func(_id: String) -> void: visible = true)
	Dialogue.ended.connect(_on_ended)
	Dialogue.line_shown.connect(_on_line)
	Dialogue.choices_shown.connect(_on_choices)


func _process(delta: float) -> void:
	if not _typing:
		return
	_shown += delta * chars_per_second
	if _shown >= _body.text.length():
		_finish_typing()
	else:
		_body.visible_characters = int(_shown)


func _input(event: InputEvent) -> void:
	if not visible or not event.is_action_pressed(&"interact"):
		return
	get_viewport().set_input_as_handled()
	if _choices.get_child_count() > 0:
		var f := get_viewport().gui_get_focus_owner() as Button
		if f != null and f.get_parent() == _choices:
			f.pressed.emit()
	else:
		_step()


func _gui_input(event: InputEvent) -> void:
	var mb := event as InputEventMouseButton
	if mb != null and mb.pressed and mb.button_index == MOUSE_BUTTON_LEFT:
		accept_event()
		if _choices.get_child_count() == 0:
			_step()


func _step() -> void:
	if _typing:
		_finish_typing()
	else:
		Dialogue.advance()


func _finish_typing() -> void:
	_typing = false
	_body.visible_characters = -1
	_more.visible = _choices.get_child_count() == 0


func _on_line(speaker: String, text: String) -> void:
	_clear_choices()
	_speaker.text = speaker.to_upper()
	_speaker.visible = speaker != ""
	_body.add_theme_color_override("default_color", UiTheme.BONE if speaker != "" else Color(0.72, 0.76, 0.78))
	_body.text = text
	_body.visible_characters = 0
	_shown = 0.0
	_typing = true
	_more.visible = false


func _on_choices(options: Array[String]) -> void:
	_finish_typing()
	_clear_choices()
	_more.visible = false
	for i in options.size():
		var b := Button.new()
		b.text = "›  " + options[i]
		b.alignment = HORIZONTAL_ALIGNMENT_LEFT
		b.pressed.connect(Dialogue.choose.bind(i))
		_choices.add_child(b)
	if not options.is_empty():
		(_choices.get_child(0) as Button).grab_focus.call_deferred()


func _on_ended(_id: String) -> void:
	_clear_choices()
	_typing = false
	visible = false


func _clear_choices() -> void:
	for c in _choices.get_children():
		_choices.remove_child(c)
		c.queue_free()

	if not _toasting:
		_next_toast()

	var t: Interactable = player.target if is_instance_valid(player) else null

	var tw := create_tween()
	tw.tween_property(_toast, "modulate:a", 1.0, 0.25)
	tw.tween_interval(2.4)
	tw.tween_property(_toast, "modulate:a", 0.0, 0.5)
	tw.tween_callback(_next_toast)

func _apply_safe_area() -> void:
	if not OS.has_feature("mobile"):
		return
	var win := Vector2(DisplayServer.window_get_size())
	if win.x <= 0.0 or win.y <= 0.0:
		return
	var safe := Rect2(DisplayServer.get_display_safe_area())
	var k := get_viewport_rect().size / win
	_safe.offset_left = safe.position.x * k.x
	_safe.offset_top = safe.position.y * k.y
	_safe.offset_right = -(win.x - safe.end.x) * k.x
	_safe.offset_bottom = -(win.y - safe.end.y) * k.y

	var dim := UiTheme.DIM.to_html(false)
	var q := ""
	for id: String in Journal.quests:
		var def: Dictionary = Journal.quest_defs.get(id, {})
		q += "[color=#%s]%s[/color]" % [amber, Journal.quest_title(id)]
		if Journal.is_complete(id):
			q += "  [color=#%s](to be continued)[/color]" % dim
		q += "\n[color=#%s]%s[/color]\n\n" % [dim, str(def.get("summary", ""))]
		for o: Dictionary in def.get("objectives", []):
			var st := Journal.objective_state(id, str(o["id"]))
			if st == Journal.DONE:

				q += "◦  %s\n" % str(o["text"])
		q += "\n"
	_quests.text = q if q != "" else "[color=#%s]Nothing yet.[/color]" % dim
	var c := ""
	for id in Journal.clues:
		var def: Dictionary = Journal.clue_defs.get(id, {})
		c += "[color=#%s]%s[/color]\n%s\n\n" % [amber, str(def.get("title", id)), str(def.get("text", ""))]

class_name TouchControls
extends Control
## Touch (and mouse) movement without on-screen buttons: drag anywhere to steer,
## tap to interact. Sits underneath every other control, so HUD chips and panels
## take their own touches first.

signal tapped

@export var deadzone := 5.0
@export var stick_radius := 22.0
@export var tap_ms := 350

var vector := Vector2.ZERO

var _down := false
var _dragged := false
var _origin := Vector2.ZERO
var _knob := Vector2.ZERO
var _t0 := 0


func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_STOP
	Dialogue.started.connect(func(_id: String) -> void: release())


func release() -> void:
	_down = false
	_dragged = false
	vector = Vector2.ZERO
	queue_redraw()


func _notification(what: int) -> void:
	if what == NOTIFICATION_PAUSED or what == NOTIFICATION_VISIBILITY_CHANGED:
		release()


func _gui_input(event: InputEvent) -> void:
	var mb := event as InputEventMouseButton
	if mb != null and mb.button_index == MOUSE_BUTTON_LEFT:
		if mb.pressed:
			_down = true
			_dragged = false
			_origin = mb.position
			_knob = mb.position
			_t0 = Time.get_ticks_msec()
		else:
			var was_tap := _down and not _dragged and Time.get_ticks_msec() - _t0 <= tap_ms
			release()
			if was_tap:
				tapped.emit()
		return
	var mm := event as InputEventMouseMotion
	if mm != null and _down:
		var d := mm.position - _origin
		if d.length() > deadzone:
			_dragged = true
		if _dragged:
			# floating stick: the origin trails the finger once it leaves the ring
			if d.length() > stick_radius:
				_origin = mm.position - d.normalized() * stick_radius
				d = mm.position - _origin
			vector = d / stick_radius
			_knob = mm.position
			queue_redraw()


func _draw() -> void:
	if not _dragged:
		return
	draw_arc(_origin, stick_radius, 0.0, TAU, 40, Color(1, 1, 1, 0.14), 1.0)
	draw_circle(_knob, 4.0, Color(1, 1, 1, 0.22))

## Interface look, built in code so nothing falls back to default Godot styling.
## PROVISIONAL: system typeface. A licensed display/body face should replace it.

	var s := StyleBoxFlat.new()
	s.bg_color = bg
	s.border_color = border
	s.set_border_width_all(border_w)

static func build() -> Theme:
	var t := Theme.new()

	f.font_names = PackedStringArray(["Avenir Next Condensed", "Avenir Next", "Helvetica Neue", "sans-serif"])

	focus.border_width_left = 2

	t.set_stylebox("hover", "Button", hover)
	t.set_stylebox("pressed", "Button", hover)
	t.set_stylebox("focus", "Button", focus)
	t.set_stylebox("disabled", "Button", flat)

	t.set_color("font_hover_color", "Button", AMBER)

## A small bordered button for HUD chips.
static func chip(button: Button) -> void:

	button.add_theme_stylebox_override("hover", box(INK, Color(AMBER, 0.8), 1, 4.0))
