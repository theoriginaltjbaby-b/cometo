@tool
class_name CharSprite
extends Node2D
## PROVISIONAL ART. A tiny person drawn in code, origin at the feet.
## To be replaced by authored sprite sheets; keep set_motion() as the interface.

@export var coat := Color(0.35, 0.4, 0.45):
	set(v):
		coat = v
		queue_redraw()
@export var accent := Color(0.7, 0.45, 0.2):
	set(v):
		accent = v
		queue_redraw()
@export var skin := Color(0.78, 0.6, 0.48):
	set(v):
		skin = v
		queue_redraw()
@export var hood := false:
	set(v):
		hood = v
		queue_redraw()

var _dir := 0  # 0 down, 1 up, 2 left, 3 right
var _moving := false
var _t := 0.0
var _frame := 0


func set_motion(facing: Vector2, moving: bool) -> void:
	var d := _dir
	if facing != Vector2.ZERO:
		if absf(facing.x) > absf(facing.y):
			d = 2 if facing.x < 0.0 else 3
		else:
			d = 1 if facing.y < 0.0 else 0
	if d != _dir or moving != _moving:
		_dir = d
		_moving = moving
		if not moving:
			_frame = 0
			_t = 0.0
		queue_redraw()


func _process(delta: float) -> void:
	if Engine.is_editor_hint() or not _moving:
		return
	_t += delta
	var f := int(_t * 8.0) % 4
	if f != _frame:
		_frame = f
		queue_redraw()


func _draw() -> void:
	var bob := -1.0 if (_moving and _frame % 2 == 1) else 0.0
	var boots := Color(0.1, 0.1, 0.12)
	var dark := coat.darkened(0.22)
	var side := _dir >= 2
	draw_set_transform(Vector2.ZERO, 0.0, Vector2(1.0, 0.45))
	draw_circle(Vector2.ZERO, 6.0, Color(0, 0, 0, 0.3))
	draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)
	# legs
	var lift_a := 1.0 if (_moving and _frame == 1) else 0.0
	var lift_b := 1.0 if (_moving and _frame == 3) else 0.0
	if side:
		draw_rect(Rect2(-2 - lift_a * 2.0, -5, 2, 5 - lift_a), boots)
		draw_rect(Rect2(0 + lift_b * 2.0, -5, 2, 5 - lift_b), boots)
	else:
		draw_rect(Rect2(-3, -5, 2, 5 - lift_a), boots)
		draw_rect(Rect2(1, -5, 2, 5 - lift_b), boots)
	# coat
	var bw := 6.0 if side else 8.0
	draw_rect(Rect2(-bw * 0.5, -15 + bob, bw, 11), coat)
	draw_rect(Rect2(-bw * 0.5, -5 + bob, bw, 1), dark)
	draw_rect(Rect2(-bw * 0.5, -11 + bob, bw, 1), accent)
	if side:
		var swing := 0.0
		if _moving:
			swing = 1.0 if _frame < 2 else -1.0
		draw_rect(Rect2(-1 + swing, -14 + bob, 2, 6), dark)
	else:
		draw_rect(Rect2(-5, -14 + bob, 1, 6), dark)
		draw_rect(Rect2(4, -14 + bob, 1, 6), dark)
	# head
	var top := coat if hood else Color(0.14, 0.11, 0.1)
	draw_rect(Rect2(-3, -21 + bob, 6, 6), skin)
	draw_rect(Rect2(-4, -22 + bob, 8, 3), top)
	var eye := Color(0.08, 0.07, 0.08)
	match _dir:
		0:
			if hood:
				draw_rect(Rect2(-4, -20 + bob, 1, 5), top)
				draw_rect(Rect2(3, -20 + bob, 1, 5), top)
			draw_rect(Rect2(-2, -18 + bob, 1, 1), eye)
			draw_rect(Rect2(1, -18 + bob, 1, 1), eye)
		1:
			draw_rect(Rect2(-4, -22 + bob, 8, 7), top)
		2:
			draw_rect(Rect2(0, -22 + bob, 4, 7), top)
			draw_rect(Rect2(-3, -18 + bob, 1, 1), eye)
		3:
			draw_rect(Rect2(-4, -22 + bob, 4, 7), top)
			draw_rect(Rect2(2, -18 + bob, 1, 1), eye)

class_name Figure3D
extends Node3D
## PROVISIONAL ART. A person built from primitive meshes, about 1.65 m and four
## heads tall, facing -Z. Procedural walk and idle animation through limb pivots.
## Keep set_motion() as the interface when an authored, rigged model replaces it.

var coat := Color(0.35, 0.4, 0.45)
var accent := Color(0.7, 0.45, 0.2)
var skin := Color(0.78, 0.6, 0.48)
var hood := false
## Draw a faint silhouette wherever the figure is hidden behind scenery (player only).
var xray := false


var _hips: Node3D
var _leg_l: Node3D
var _leg_r: Node3D
var _arm_l: Node3D
var _arm_r: Node3D
var _torso: Node3D
var _phase := 0.0
var _amount := 0.0
var _speed := 0.0
var _t := 0.0


func _ready() -> void:
	var coat_m := _mat(coat, 0.85)
	var dark_m := _mat(coat.darkened(0.3), 0.9)
	var skin_m := _mat(skin, 0.7)
	var boot_m := _mat(Color(0.1, 0.09, 0.09), 0.6)
	var accent_m := _mat(accent, 0.6, 0.3)
	_hips = Node3D.new()
	_hips.position.y = 0.82
	add_child(_hips)
	_leg_l = _limb(_hips, Vector3(-0.1, 0, 0), 0.075, 0.8, dark_m, boot_m)
	_leg_r = _limb(_hips, Vector3(0.1, 0, 0), 0.075, 0.8, dark_m, boot_m)
	_torso = Node3D.new()
	_hips.add_child(_torso)
	# coat skirt, chest, belt
	var skirt := CylinderMesh.new()
	skirt.top_radius = 0.19
	skirt.bottom_radius = 0.27
	skirt.height = 0.5
	skirt.radial_segments = 12
	_part(_torso, skirt, coat_m, Vector3(0, -0.05, 0.01), Vector3(1, 1, 0.85))
	var hem := CylinderMesh.new()
	hem.top_radius = 0.272
	hem.bottom_radius = 0.272
	hem.height = 0.04
	hem.radial_segments = 12
	_part(_torso, hem, dark_m, Vector3(0, -0.29, 0.01), Vector3(1, 1, 0.85))
	var chest := CapsuleMesh.new()
	chest.radius = 0.19
	chest.height = 0.6
	_part(_torso, chest, coat_m, Vector3(0, 0.36, 0), Vector3(1, 1, 0.78))
	var belt := CylinderMesh.new()
	belt.top_radius = 0.2
	belt.bottom_radius = 0.2
	belt.height = 0.05
	_part(_torso, belt, accent_m, Vector3(0, 0.18, 0))
	var collar := CylinderMesh.new()
	collar.top_radius = 0.11
	collar.bottom_radius = 0.17
	collar.height = 0.1
	_part(_torso, collar, accent_m, Vector3(0, 0.6, 0))
	# shoulders, scarf, satchel and strap
	var sh := SphereMesh.new()
	sh.radius = 0.1
	sh.height = 0.16
	_part(_torso, sh, coat_m, Vector3(-0.21, 0.55, 0))
	_part(_torso, sh, coat_m, Vector3(0.21, 0.55, 0))
	var scarf := BoxMesh.new()
	scarf.size = Vector3(0.09, 0.26, 0.04)
	_part(_torso, scarf, accent_m, Vector3(0.06, 0.45, -0.165), Vector3.ONE).rotation.z = 0.12
	var bag_m := _mat(Color(0.3, 0.2, 0.13), 0.8)
	var bag := BoxMesh.new()
	bag.size = Vector3(0.08, 0.2, 0.24)
	_part(_torso, bag, bag_m, Vector3(0.27, 0.06, 0.02))
	var strap := BoxMesh.new()
	strap.size = Vector3(0.04, 0.62, 0.02)
	var st := _part(_torso, strap, bag_m, Vector3(0.03, 0.35, -0.155))
	st.rotation.z = -0.62
	var st2 := _part(_torso, strap, bag_m, Vector3(0.03, 0.35, 0.155))
	st2.rotation.z = -0.62
	_arm_l = _limb(_torso, Vector3(-0.25, 0.53, 0), 0.06, 0.56, coat_m, skin_m)
	_arm_r = _limb(_torso, Vector3(0.25, 0.53, 0), 0.06, 0.56, coat_m, skin_m)
	# head
	var head := SphereMesh.new()
	head.radius = 0.19
	head.height = 0.4
	_part(_torso, head, skin_m, Vector3(0, 0.81, 0))
	var eye := BoxMesh.new()
	eye.size = Vector3(0.035, 0.035, 0.02)
	var eye_m := _mat(Color(0.06, 0.05, 0.06), 0.5)
	_part(_torso, eye, eye_m, Vector3(-0.065, 0.83, -0.175))
	_part(_torso, eye, eye_m, Vector3(0.065, 0.83, -0.175))
	if hood:
		var h := SphereMesh.new()
		h.radius = 0.22
		h.height = 0.46
		_part(_torso, h, coat_m, Vector3(0, 0.84, 0.035), Vector3(1, 1, 0.95))
		var face := SphereMesh.new()
		face.radius = 0.17
		face.height = 0.34
		_part(_torso, face, skin_m, Vector3(0, 0.8, -0.05))
		var peak := BoxMesh.new()
		peak.size = Vector3(0.36, 0.05, 0.1)
		_part(_torso, peak, coat_m, Vector3(0, 0.98, -0.17)).rotation.x = 0.35
	else:
		var hair := SphereMesh.new()
		hair.radius = 0.2
		hair.height = 0.26
		hair.is_hemisphere = true
		_part(_torso, hair, _mat(Color(0.12, 0.09, 0.08), 0.8), Vector3(0, 0.85, 0.02))


func set_motion(moving: bool, speed_ratio: float = 1.0) -> void:
	_speed = clampf(speed_ratio, 0.0, 1.2) if moving else 0.0


func _process(delta: float) -> void:
	_t += delta
	_amount = move_toward(_amount, 1.0 if _speed > 0.05 else 0.0, delta * 6.0)
	_phase += delta * 10.0 * maxf(_speed, 0.4) * _amount
	var s := sin(_phase) * _amount
	_leg_l.rotation.x = s * 0.6
	_leg_r.rotation.x = -s * 0.6
	_arm_l.rotation.x = -s * 0.5
	_arm_r.rotation.x = s * 0.5
	_arm_l.rotation.z = -0.08
	_arm_r.rotation.z = 0.08
	var breathe := sin(_t * 2.0) * 0.012 * (1.0 - _amount)
	_hips.position.y = 0.82 + absf(cos(_phase)) * 0.04 * _amount
	_torso.rotation.x = -0.08 * _amount
	_torso.scale = Vector3(1.0, 1.0 + breathe, 1.0)


func _limb(parent: Node3D, at: Vector3, r: float, length: float, m: Material, end_m: Material) -> Node3D:
	var pivot := Node3D.new()
	pivot.position = at
	parent.add_child(pivot)
	var c := CylinderMesh.new()
	c.top_radius = r * 1.15
	c.bottom_radius = r * 0.85
	c.height = length - r
	c.radial_segments = 8
	_part(pivot, c, m, Vector3(0, -length * 0.5 + r * 0.25, 0))
	var joint := SphereMesh.new()
	joint.radius = r * 1.15
	joint.height = r * 2.3
	_part(pivot, joint, m, Vector3.ZERO)
	if end_m.get_meta("boot", false):
		var b := BoxMesh.new()
		b.size = Vector3(r * 2.2, r * 1.4, r * 3.6)
		_part(pivot, b, end_m, Vector3(0, -length + r * 0.5, -r * 0.7))
		var cuff := CylinderMesh.new()
		cuff.top_radius = r * 1.25
		cuff.bottom_radius = r * 1.15
		cuff.height = r * 2.6
		cuff.radial_segments = 8
		_part(pivot, cuff, end_m, Vector3(0, -length + r * 1.9, 0))
	else:
		var sp := SphereMesh.new()
		sp.radius = r * 1.05
		sp.height = r * 2.1
		_part(pivot, sp, end_m, Vector3(0, -length + r * 0.6, 0))
	return pivot


func _part(parent: Node3D, mesh: Mesh, m: Material, at: Vector3, s: Vector3 = Vector3.ONE) -> MeshInstance3D:
	var mi := MeshInstance3D.new()
	mi.mesh = mesh
	mi.material_override = m
	mi.position = at
	mi.scale = s
	parent.add_child(mi)
	return mi


func _mat(c: Color, rough: float, metal: float = 0.0) -> StandardMaterial3D:
	var m := StandardMaterial3D.new()
	m.albedo_color = c
	m.roughness = rough
	m.metallic = metal
	if xray:
		m.stencil_mode = BaseMaterial3D.STENCIL_MODE_XRAY
		m.stencil_color = Color(1.0, 0.72, 0.35, 0.5)
	if c.v < 0.12:
		m.set_meta("boot", true)
	return m

shader_type canvas_item;
// PROVISIONAL. Drifting sea mist plus a soft vignette. One layer, no screen read.

uniform vec4 fog_color : source_color = vec4(0.6, 0.69, 0.75, 1.0);
uniform float fog_amount = 0.11;
uniform float vignette = 0.5;

float hash(vec2 p) {
	return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float noise(vec2 p) {
	vec2 i = floor(p);
	vec2 f = fract(p);
	f = f * f * (3.0 - 2.0 * f);
	return mix(mix(hash(i), hash(i + vec2(1.0, 0.0)), f.x),
			mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), f.x), f.y);
}

void fragment() {
	float n = noise(UV * vec2(3.0, 2.0) + vec2(TIME * 0.02, 0.0)) * 0.6
			+ noise(UV * vec2(7.0, 4.0) - vec2(TIME * 0.035, 0.0)) * 0.4;
	float fog = fog_amount * n * 1.6;
	float v = smoothstep(0.35, 0.85, distance(UV, vec2(0.5))) * vignette;
	vec3 col = mix(fog_color.rgb, vec3(0.01, 0.02, 0.04), v / (v + fog + 0.0001));
	COLOR = vec4(col, clamp(fog + v, 0.0, 0.9));
}

@tool
class_name ProvBlock
extends StaticBody2D
## PROVISIONAL ART. A solid piece of the world drawn in code, in 3/4 view.
## The node origin is the bottom-centre of the footprint (its front edge), so
## y-sorting works. `size` is the footprint (width x depth); `height` is how far
## the front face rises. Replace with authored art; keep the footprints.

@export var size := Vector2(48, 24):
	set(v):
		size = v
		queue_redraw()
@export var height := 20.0:
	set(v):
		height = v
		queue_redraw()
@export_enum("pod", "hut", "rock", "fence", "gate", "crate", "lamp", "tree", "pit", "adit", "sign", "none") var style := "crate":
	set(v):
		style = v
		queue_redraw()
@export var lit := false:
	set(v):
		lit = v
		queue_redraw()
@export var solid := true
@export var tint := Color(1, 1, 1):
	set(v):
		tint = v
		queue_redraw()

static var _light_tex: GradientTexture2D

var _shape: CollisionShape2D
var _solid_now := true


func _ready() -> void:
	if Engine.is_editor_hint():
		return
	collision_layer = 1
	collision_mask = 0
	if solid:
		var rect := RectangleShape2D.new()
		rect.size = size
		_shape = CollisionShape2D.new()
		_shape.shape = rect
		_shape.position = Vector2(0, -size.y * 0.5)
		_shape.disabled = not _solid_now
		add_child(_shape)
	if lit:
		_add_light()


func set_solid(v: bool) -> void:
	_solid_now = v
	if _shape != null:
		_shape.set_deferred("disabled", not v)


static func light_texture() -> GradientTexture2D:
	if _light_tex == null:
		var g := Gradient.new()
		g.set_color(0, Color(1, 1, 1, 1))
		g.set_color(1, Color(0, 0, 0, 0))
		_light_tex = GradientTexture2D.new()
		_light_tex.gradient = g
		_light_tex.width = 128
		_light_tex.height = 128
		_light_tex.fill = GradientTexture2D.FILL_RADIAL
		_light_tex.fill_from = Vector2(0.5, 0.5)
		_light_tex.fill_to = Vector2(1.0, 0.5)
	return _light_tex


func _add_light() -> void:
	var l := PointLight2D.new()
	l.texture = light_texture()
	l.color = Color(1.0, 0.6, 0.3)
	l.energy = 1.1
	if style == "lamp":
		l.position = Vector2(3, -height + 4)
		l.texture_scale = 1.9
	else:
		l.position = Vector2(size.x * 0.5 - 15, -height * 0.5 + 6)
		l.texture_scale = 1.2
	add_child(l)


func _draw() -> void:
	var w := size.x
	var d := size.y
	var h := height
	var l := -w * 0.5
	var rng := RandomNumberGenerator.new()
	rng.seed = hash(Vector2i(position))
	match style:
		"pod", "hut":
			var wall := (Color(0.4, 0.45, 0.48) if style == "pod" else Color(0.3, 0.36, 0.33)) * tint
			var roof := wall.lightened(0.14)
			draw_rect(Rect2(l, -d - h, w, d), roof)
			for i in range(1, int(w / 8.0)):
				draw_line(Vector2(l + i * 8, -d - h), Vector2(l + i * 8, -h), roof.darkened(0.14), 1.0)
			draw_rect(Rect2(l, -h, w, h), wall)
			draw_rect(Rect2(l, -h, w, 1), wall.darkened(0.35))
			draw_rect(Rect2(l, -2, w, 2), wall.darkened(0.4))
			draw_rect(Rect2(l, -h + 3, w, 1), Color(0.72, 0.4, 0.2, 0.75))
			draw_rect(Rect2(l + 6, -13, 8, 13), wall.darkened(0.5))
			draw_rect(Rect2(l + w - 23, -h + 6, 15, 8), wall.darkened(0.5))
			draw_rect(Rect2(l + w - 22, -h + 7, 13, 6), Color(1.0, 0.8, 0.45) if lit else Color(0.1, 0.14, 0.18))
			for i in 3:
				var x := l + rng.randf() * w
				draw_line(Vector2(x, -h + 4), Vector2(x, -h + 4 + rng.randf_range(4, 12)), Color(0.45, 0.27, 0.18, 0.5), 1.0)
		"rock":
			var face := Color(0.19, 0.22, 0.25) * tint
			draw_rect(Rect2(l, -d - h, w, d), Color(0.24, 0.28, 0.29) * tint)

## PROVISIONAL ART. Narrow-gauge rail drawn along `points`. Also the path data
## that OreCart follows.

@export var points := PackedVector2Array():
	set(v):
		points = v
		queue_redraw()
@export var buffer_start := false
@export var buffer_end := false


func _draw() -> void:
	var tie := Color(0.2, 0.16, 0.13)
	var steel := Color(0.42, 0.43, 0.45)
	for i in points.size() - 1:
		var a := points[i]
		var b := points[i + 1]
		var dir := (b - a).normalized()
		var n := dir.orthogonal()
		var length := a.distance_to(b)
		var s := 0.0
		while s < length:
			var p := a + dir * s
			draw_line(p - n * 6.0, p + n * 6.0, tie, 2.0)
			s += 6.0
		draw_line(a - n * 4.0, b - n * 4.0, steel, 1.0)
		draw_line(a + n * 4.0, b + n * 4.0, steel, 1.0)
	if buffer_start and points.size() > 1:
		_buffer(points[0], (points[0] - points[1]).normalized())
	if buffer_end and points.size() > 1:
		var e := points.size() - 1
		_buffer(points[e], (points[e] - points[e - 1]).normalized())


func _buffer(at: Vector2, out: Vector2) -> void:
	var p := at + out * 12.0
	var n := out.orthogonal()
	draw_line(p - n * 7.0, p + n * 7.0, Color(0.5, 0.2, 0.15), 3.0)

		draw_line(_pos[i], _pos[i] + fall * _speed[i] * streak, Color(0.75, 0.84, 0.92, 0.22), 1.0)

== core/dialogue.gd
==============================================================================
extends Node
## Data-driven dialogue runner. UI listens to the signals; nothing here draws.
##
## res://data/dialogue/<id>.json:
## {
##   "entry": [ {"if": cond, "goto": "node"}, {"goto": "first"} ],
##   "nodes": {
##     "first": {
##       "do":   [effects on entering the node],
##       "say":  [["Speaker", "line"], ["", "narration"]],
##       "then": [effects once the lines have been read],
##       "choices": [ {"text": "...", "if": cond, "do": [effects], "goto": "node"} ],
##       "goto": "node"            (used when no choice is available; default "end")
##     }
##   }
## }

signal started(id: String)
signal line_shown(speaker: String, text: String)
signal choices_shown(options: Array[String])
signal ended(id: String)

const ROOT := "res://data/dialogue/%s.json"

var active := false
var current_id := ""
var current_choices: Array[Dictionary] = []

var _data: Dictionary = {}
var _node: Dictionary = {}
var _lines: Array = []
var _line_i := 0
var _cache: Dictionary = {}
var _hops := 0


func start(id: String) -> bool:
	if active:
		return false
	if not _cache.has(id):
		_cache[id] = Data.load_dict(ROOT % id)
	_data = _cache[id]
	if _data.is_empty():
		push_warning("Dialogue: nothing to say for '%s'" % id)
		return false
	current_id = id
	active = true
	started.emit(id)
	var entry := "start"
	for e: Dictionary in _data.get("entry", []):
		if GameState.check(e.get("if")):
			entry = str(e.get("goto", "start"))
			break
	_hops = 0
	_enter(entry)
	return true


func advance() -> void:
	if not active or not current_choices.is_empty():
		return
	_line_i += 1
	if _line_i < _lines.size():
		_emit_line()
	else:
		_after_lines()


func choose(index: int) -> void:
	if not active or index < 0 or index >= current_choices.size():
		return
	var c := current_choices[index]
	current_choices = []
	GameState.apply(c.get("do"))
	_hops = 0
	_enter(str(c.get("goto", "end")))


func cancel() -> void:
	if active:
		_finish()


func choice_texts() -> Array[String]:
	var out: Array[String] = []
	for c in current_choices:
		out.append(str(c.get("text", "...")))
	return out


func _enter(node_name: String) -> void:
	var nodes: Dictionary = _data.get("nodes", {})
	_hops += 1
	if node_name == "end" or not nodes.has(node_name) or _hops > 32:
		if node_name != "end":
			push_warning("Dialogue '%s': bad node '%s'" % [current_id, node_name])
		_finish()
		return
	_node = nodes[node_name]
	GameState.apply(_node.get("do"))
	_lines = _node.get("say", [])
	_line_i = 0
	if _lines.is_empty():
		_after_lines()
	else:
		_emit_line()


func _emit_line() -> void:
	var l: Array = _lines[_line_i]
	line_shown.emit(str(l[0]), str(l[1]))


func _after_lines() -> void:
	GameState.apply(_node.get("then"))
	var opts: Array[Dictionary] = []
	for c: Dictionary in _node.get("choices", []):
		if GameState.check(c.get("if")):
			opts.append(c)
	if opts.is_empty():
		_enter(str(_node.get("goto", "end")))
	else:
		current_choices = opts
		choices_shown.emit(choice_texts())


func _finish() -> void:
	var id := current_id
	active = false
	current_id = ""
	current_choices = []
	_lines = []
	ended.emit(id)

== core/game_state.gd

extends Node
## Persistent world state: a flat dictionary of flags, plus the tiny condition/effect
## language shared by dialogue, interactables and puzzles.
##
## Conditions (Dictionary):
##   {"flag": "k"}                truthy flag
##   {"flag": "k", "eq": value}   flag equals value
##   {"clue": "id"}               journal holds the clue
##   {"not": cond} {"all": [..]} {"any": [..]}
## Effects (Array of Dictionary):
##   {"set": {"k": value}}  {"clue": "id"}  {"quest": "id"}
##   {"reveal": "quest.objective"}  {"objective": "quest.objective"}  {"complete": "quest"}
##   {"notify": "text"}

signal flag_changed(key: String, value: Variant)
signal notified(text: String)
## Asks the world (game/main.gd) to move the player to another map.
signal travel_requested(map_id: String, spawn: String)
## A named place was entered; the HUD shows it as a title card.
signal location_entered(text: String)

var flags: Dictionary = {}


func _ready() -> void:
	InputSetup.ensure()


func reset() -> void:
	flags = {}


func get_flag(key: String, default: Variant = false) -> Variant:
	return flags.get(key, default)


func set_flag(key: String, value: Variant) -> void:
	if flags.has(key) and _same(flags[key], value):
		return
	flags[key] = value
	flag_changed.emit(key, value)


func notify(text: String) -> void:
	notified.emit(text)


func check(cond: Variant) -> bool:
	if cond == null:
		return true
	if cond is Array:
		for c: Variant in cond:
			if not check(c):
				return false
		return true
	if not cond is Dictionary:
		return bool(cond)
	var d: Dictionary = cond
	if d.is_empty():
		return true
	if d.has("all"):
		return check(d["all"])
	if d.has("any"):
		for c: Variant in d["any"]:
			if check(c):
				return true
		return false
	if d.has("not"):
		return not check(d["not"])
	if d.has("clue"):
		return Journal.has_clue(str(d["clue"]))
	if d.has("flag"):
		var v: Variant = get_flag(str(d["flag"]))
		if d.has("eq"):
			return _same(v, d["eq"])
		return _truthy(v)
	push_warning("GameState.check: unknown condition %s" % [d])
	return true


func apply(effects: Variant) -> void:
	if not effects is Array:
		return
	for e: Variant in effects:
		if not e is Dictionary:
			continue
		var d: Dictionary = e
		if d.has("set"):
			var s: Dictionary = d["set"]
			for k: Variant in s:
				set_flag(str(k), s[k])
		if d.has("clue"):
			Journal.add_clue(str(d["clue"]))
		if d.has("quest"):
			Journal.start_quest(str(d["quest"]))
		if d.has("reveal"):
			Journal.reveal_objective(str(d["reveal"]))
		if d.has("objective"):
			Journal.complete_objective(str(d["objective"]))
		if d.has("complete"):
			Journal.complete_quest(str(d["complete"]))
		if d.has("notify"):
			notify(str(d["notify"]))


func to_dict() -> Dictionary:
	return flags.duplicate(true)


func from_dict(d: Dictionary) -> void:
	flags = d.duplicate(true)


static func _is_num(v: Variant) -> bool:
	return typeof(v) == TYPE_INT or typeof(v) == TYPE_FLOAT


static func _same(a: Variant, b: Variant) -> bool:
	if _is_num(a) and _is_num(b):
		return is_equal_approx(float(a), float(b))
	return typeof(a) == typeof(b) and a == b


static func _truthy(v: Variant) -> bool:
	match typeof(v):
		TYPE_NIL:
			return false
		TYPE_BOOL:
			return v
		TYPE_INT, TYPE_FLOAT:
			return v != 0
		TYPE_STRING:
			return v != ""
	return true

extends Node
## Investigations (quests with objectives) and clues. Definitions are data
## (res://data/journal); this node only holds the player's progress.

signal updated(kind: String, text: String)

const HIDDEN := "hidden"
const ACTIVE := "active"
const DONE := "done"

var quest_defs: Dictionary = {}
var clue_defs: Dictionary = {}
## quest id -> {"objectives": {objective id: state}, "complete": bool}
var quests: Dictionary = {}
var clues: Array[String] = []


func _ready() -> void:
	quest_defs = Data.load_dict("res://data/journal/quests.json")
	clue_defs = Data.load_dict("res://data/journal/clues.json")


func reset() -> void:
	quests = {}
	clues.clear()


func start_quest(id: String) -> void:
	if quests.has(id) or not quest_defs.has(id):
		return
	var states := {}
	for o: Dictionary in quest_defs[id].get("objectives", []):
		states[str(o["id"])] = HIDDEN if o.get("hidden", false) else ACTIVE
	quests[id] = {"objectives": states, "complete": false}
	updated.emit("quest", quest_title(id))


func reveal_objective(path: String) -> void:
	var p := _split(path)
	if p.is_empty() or _state(p[0], p[1]) != HIDDEN:
		return
	quests[p[0]]["objectives"][p[1]] = ACTIVE
	updated.emit("objective_new", objective_text(p[0], p[1]))


func complete_objective(path: String) -> void:
	var p := _split(path)
	if p.is_empty() or _state(p[0], p[1]) in ["", DONE]:
		return
	quests[p[0]]["objectives"][p[1]] = DONE
	updated.emit("objective", objective_text(p[0], p[1]))


func complete_quest(id: String) -> void:
	if not quests.has(id) or quests[id]["complete"]:
		return
	quests[id]["complete"] = true
	updated.emit("complete", quest_title(id))


func add_clue(id: String) -> void:
	if clues.has(id) or not clue_defs.has(id):
		return
	clues.append(id)
	updated.emit("clue", str(clue_defs[id].get("title", id)))


func has_clue(id: String) -> bool:
	return clues.has(id)


func objective_state(quest: String, objective: String) -> String:
	return _state(quest, objective)


func is_complete(quest: String) -> bool:
	return quests.has(quest) and quests[quest]["complete"]


func quest_title(id: String) -> String:
	return str(quest_defs.get(id, {}).get("title", id))


func objective_text(quest: String, objective: String) -> String:
	for o: Dictionary in quest_defs.get(quest, {}).get("objectives", []):
		if str(o["id"]) == objective:
			return str(o.get("text", objective))
	return objective


func to_dict() -> Dictionary:
	return {"quests": quests.duplicate(true), "clues": clues.duplicate()}


func from_dict(d: Dictionary) -> void:
	quests = (d.get("quests", {}) as Dictionary).duplicate(true)
	clues.clear()
	for c: Variant in d.get("clues", []):
		clues.append(str(c))


func _split(path: String) -> PackedStringArray:
	var p := path.split(".")
	if p.size() != 2 or not quests.has(p[0]):
		return PackedStringArray()
	return p


func _state(quest: String, objective: String) -> String:
	if not quests.has(quest):
		return ""
	return str(quests[quest]["objectives"].get(objective, ""))

		"world": world.call("capture_world"),
	}
	var path := slot_path(slot)
	var tmp := path + ".tmp"
	var f := FileAccess.open(tmp, FileAccess.WRITE)
	if f == null:
		push_error("SaveManager: cannot write %s" % tmp)
		return false
	f.store_string(JSON.stringify(data, "\t"))
	f.close()
	if FileAccess.file_exists(path):
		DirAccess.remove_absolute(path)
	if DirAccess.rename_absolute(tmp, path) != OK:
		push_error("SaveManager: cannot finalise %s" % path)
		return false
	saved.emit(slot)
	return true


func load_game(slot: String = "manual") -> bool:
	if world == null or not has_save(slot):
		return false
	var data: Dictionary = Data.load_dict(slot_path(slot))
	if data.is_empty() or int(data.get("version", 0)) != VERSION:
		push_warning("SaveManager: save '%s' is unreadable or from another version" % slot)
		return false
	GameState.from_dict(data.get("flags", {}))
	Journal.from_dict(data.get("journal", {}))
	world.call("restore_world", data.get("world", {}))
	loaded.emit(slot)
	return true


func delete_save(slot: String) -> void:
	if has_save(slot):
		DirAccess.remove_absolute(slot_path(slot))


func _notification(what: int) -> void:
	if what == NOTIFICATION_APPLICATION_PAUSED or what == NOTIFICATION_WM_CLOSE_REQUEST \
			or what == NOTIFICATION_WM_GO_BACK_REQUEST:
		if autosave_enabled and world != null:
			save_game("auto")

extends Node
## Boots the game, owns the current map and the player, and is the "world" that
## SaveManager captures and restores.

const MAPS := {
	"anjabi_mine": "res://maps/anjabi/mine_3d.tscn",
	"anjabi_town": "res://maps/anjabi/town_3d.tscn",
}
const START_MAP := "anjabi_town"
const START_QUEST := "quiet_mine"
const PLAYER_SCENE := preload("res://game/player/player.tscn")

## GameMap (2D) or GameMap3D. The player is Player or Player3D to match.
var map
var player
var map_id := ""

@onready var world: Node2D = $World
@onready var ui: GameUi = $UI


func _ready() -> void:
	SaveManager.world = self
	ui.new_game_requested.connect(new_game)
	ui.touch.tapped.connect(_on_tapped)
	GameState.travel_requested.connect(_on_travel_requested)
	if SaveManager.boot_from_autosave and SaveManager.has_save("auto") and SaveManager.load_game("auto"):
		return
	new_game()


func new_game() -> void:
	GameState.reset()
	Journal.reset()
	load_map(START_MAP, "default")
	Journal.start_quest(START_QUEST)


func load_map(id: String, spawn: String = "default") -> void:
	_enter_map(id)
	player.global_position = map.spawn_position(spawn)
	if player is Player and player.walk_mask != null:
		player.global_position = player.walk_mask.nearest_walkable(player.global_position)
	player.camera.reset_smoothing()


func capture_world() -> Dictionary:
	if player is Player3D:
		var p: Vector3 = player.global_position
		return {"map": map_id, "dim": 3, "x": p.x, "y": p.y, "z": p.z, "fx": player.facing.x, "fz": player.facing.z}
	return {
		"map": map_id,
		"x": player.global_position.x,
		"y": player.global_position.y,
		"fx": player.facing.x,
		"fy": player.facing.y,
	}


func restore_world(d: Dictionary) -> void:
	var id := str(d.get("map", START_MAP))
	var known := MAPS.has(id)
	if not known:
		# e.g. saves from the retired separate district map: resume at the Steps
		id = START_MAP
	_enter_map(id)
	if player is Player3D:
		if known and int(d.get("dim", 2)) == 3:
			player.global_position = Vector3(float(d.get("x", 0.0)), float(d.get("y", 0.0)), float(d.get("z", 0.0)))
			var f := Vector3(float(d.get("fx", 0.0)), 0.0, float(d.get("fz", 1.0)))
			player.facing = f.normalized() if f.length() > 0.01 else Vector3(0, 0, 1)
			player.face_now()
		else:
			player.global_position = map.spawn_position("default")
	else:
		player.global_position = Vector2(float(d.get("x", 0.0)), float(d.get("y", 0.0)))
		player.facing = Vector2(float(d.get("fx", 0.0)), float(d.get("fy", 1.0)))
	player.camera.reset_smoothing()


func _enter_map(id: String) -> void:
	Dialogue.cancel()
	get_tree().paused = false
	if map != null:
		world.remove_child(map)
		map.queue_free()
	map_id = id
	map = (load(MAPS[id]) as PackedScene).instantiate()
	world.add_child(map)
	if map is GameMap3D:
		player = Player3D.new()
		map.entities.add_child(player)
		player.setup(ui.touch)
	else:
		player = PLAYER_SCENE.instantiate() as Player
		map.entities.add_child(player)
		player.setup(ui.touch, map.bounds)
		player.walk_mask = map.get_node_or_null("WalkMask") as WalkMask
	map.player = player
	ui.bind_player(player)
	# the 2D mist overlay is for 2D maps; 3D maps carry their own fog
	$Weather/Fog.visible = not (map is GameMap3D)
	ui.hud.show_location(map.display_name)


func _on_travel_requested(id: String, spawn: String) -> void:
	if MAPS.has(id):
		load_map.call_deferred(id, spawn)


func _on_tapped() -> void:
	if player != null:
		player.try_interact()

class_name FollowCamera
extends Camera3D
## Fixed-orientation elevated RPG camera. Never rotates: north is always up the
## screen. Orthographic by default; `size` is the visible height in metres.

@export var pitch_deg := 50.0
@export var distance := 40.0
@export var ortho_size := 14.0
@export var follow_speed := 6.0
## Aim slightly ahead of the feet so the player sits a little below centre.
@export var look_height := 1.0

var target: Node3D


func _ready() -> void:
	top_level = true
	projection = PROJECTION_ORTHOGONAL
	size = ortho_size
	near = 0.5
	far = 160.0
	rotation_degrees = Vector3(-pitch_deg, 0, 0)
	current = true
	reset_smoothing()


func _offset() -> Vector3:
	var p := deg_to_rad(pitch_deg)
	return Vector3(0, sin(p), cos(p)) * distance + Vector3(0, look_height, 0)


func reset_smoothing() -> void:
	if target != null and target.is_inside_tree():
		global_position = target.global_position + _offset()


func _physics_process(delta: float) -> void:
	if target == null:
		return
	var want := target.global_position + _offset()
	global_position = global_position.lerp(want, 1.0 - exp(-follow_speed * delta))

class_name Player
extends CharacterBody2D

signal target_changed(target: Interactable)

@export var walk_speed := 64.0

var touch: TouchControls
var facing := Vector2.DOWN
var target: Interactable
## Optional painted walkable area (maps without collision bodies).
var walk_mask: WalkMask

@onready var sprite: CharSprite = $Sprite
@onready var reach: Area2D = $Reach
@onready var camera: Camera2D = $Camera


func setup(touch_controls: TouchControls, bounds: Rect2) -> void:
	touch = touch_controls
	camera.limit_left = int(bounds.position.x)
	camera.limit_top = int(bounds.position.y)
	camera.limit_right = int(bounds.end.x)
	camera.limit_bottom = int(bounds.end.y)
	camera.reset_smoothing()


func _physics_process(_delta: float) -> void:
	var dir := Vector2.ZERO
	if not Dialogue.active:
		dir = Input.get_vector(&"move_left", &"move_right", &"move_up", &"move_down")
		if dir == Vector2.ZERO and touch != null:
			dir = touch.vector
	var moving := dir.length() > 0.1
	velocity = _allowed(dir.limit_length(1.0) * walk_speed, _delta) if moving else Vector2.ZERO
	move_and_slide()
	if moving:
		facing = dir.normalized()
	sprite.set_motion(facing, moving)
	_update_target()


func _allowed(v: Vector2, delta: float) -> Vector2:
	if walk_mask == null or not _clear(global_position):
		return v
	if _clear(global_position + v * delta):
		return v
	var vx := Vector2(v.x, 0.0)
	if v.x != 0.0 and _clear(global_position + vx * delta):
		return vx
	var vy := Vector2(0.0, v.y)
	if v.y != 0.0 and _clear(global_position + vy * delta):
		return vy
	return Vector2.ZERO


func _clear(p: Vector2) -> bool:
	return walk_mask.is_walkable(p) and walk_mask.is_walkable(p + Vector2(3, 0)) \
			and walk_mask.is_walkable(p + Vector2(-3, 0)) and walk_mask.is_walkable(p + Vector2(0, -3))


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed(&"interact") and try_interact():
		get_viewport().set_input_as_handled()


func try_interact() -> bool:
	if Dialogue.active or target == null or not target.is_available():
		return false
	target.interact(self)
	return true


func _update_target() -> void:
	var best: Interactable = null
	var best_d := INF
	for a in reach.get_overlapping_areas():
		var it := a as Interactable
		if it == null or not it.is_available():
			continue
		var d := global_position.distance_squared_to(it.global_position)
		if d < best_d:
			best_d = d
			best = it
	if best != target:
		target = best
		target_changed.emit(target)

class_name Player3D
extends CharacterBody3D
## The player in 3D maps. Same outward interface as Player (target, facing,
## try_interact, setup, camera.reset_smoothing) so main, HUD and tests can drive
## either. Screen-up is north (-Z); the camera never rotates.

signal target_changed(target: Interactable3D)

@export var walk_speed := 4.2
@export var accel := 28.0
@export var decel := 36.0
@export var gravity := 22.0

var touch: TouchControls
var facing := Vector3(0, 0, 1)
var target: Interactable3D
## Not used in 3D; kept so code written for 2D maps can test it safely.
var walk_mask: Variant = null
var figure: Figure3D
var reach: Area3D
var camera: FollowCamera
## Last place the player stood; used if they ever fall out of the world.
var _safe := Vector3.ZERO
var _has_safe := false


func _init() -> void:
	collision_layer = 2
	collision_mask = 1
	floor_snap_length = 0.45
	floor_max_angle = deg_to_rad(52.0)
	floor_constant_speed = true
	var cs := CollisionShape3D.new()
	var cap := CapsuleShape3D.new()
	cap.radius = 0.28
	cap.height = 1.6
	cs.shape = cap
	cs.position.y = 0.8
	add_child(cs)
	figure = Figure3D.new()
	figure.coat = Color(0.86, 0.62, 0.2)
	figure.accent = Color(0.2, 0.24, 0.28)
	figure.hood = true
	figure.xray = true
	add_child(figure)
	reach = Area3D.new()
	reach.collision_layer = 0
	reach.collision_mask = 4
	reach.monitorable = false
	var rs := CollisionShape3D.new()
	var sphere := SphereShape3D.new()
	sphere.radius = 1.4
	rs.shape = sphere
	rs.position = Vector3(0, 0.8, 0)
	reach.add_child(rs)
	add_child(reach)
	camera = FollowCamera.new()
	camera.target = self
	add_child(camera)


func setup(touch_controls: TouchControls, _bounds: Variant = null) -> void:
	touch = touch_controls
	camera.reset_smoothing()


func _physics_process(delta: float) -> void:
	var dir := Vector2.ZERO
	if not Dialogue.active:
		dir = Input.get_vector(&"move_left", &"move_right", &"move_up", &"move_down")
		if dir == Vector2.ZERO and touch != null:
			dir = touch.vector
	var moving := dir.length() > 0.1
	var want := Vector3(dir.x, 0, dir.y).limit_length(1.0) * walk_speed if moving else Vector3.ZERO
	var h := Vector3(velocity.x, 0, velocity.z).move_toward(want, (accel if moving else decel) * delta)
	velocity.x = h.x
	velocity.z = h.z
	if is_on_floor():
		velocity.y = 0.0
	else:
		velocity.y -= gravity * delta
	move_and_slide()
	if is_on_floor():
		_safe = global_position
		_has_safe = true
	elif global_position.y < -40.0 and _has_safe:
		global_position = _safe + Vector3(0, 0.3, 0)
		velocity = Vector3.ZERO
	if moving:
		facing = Vector3(dir.x, 0, dir.y).normalized()
	var yaw := atan2(-facing.x, -facing.z)
	figure.rotation.y = lerp_angle(figure.rotation.y, yaw, 1.0 - exp(-14.0 * delta))
	figure.set_motion(h.length() > 0.3, h.length() / walk_speed)
	_update_target()


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed(&"interact") and try_interact():
		get_viewport().set_input_as_handled()


func try_interact() -> bool:
	if Dialogue.active or target == null or not target.is_available():
		return false
	target.interact(self)
	return true


func face_now() -> void:
	figure.rotation.y = atan2(-facing.x, -facing.z)


func _update_target() -> void:
	var best: Interactable3D = null
	var best_d := INF
	for a in reach.get_overlapping_areas():
		var it := a as Interactable3D
		if it == null or not it.is_available():
			continue
		var d := global_position.distance_squared_to(it.global_position)
		if d < best_d:
			best_d = d
			best = it
	if best != target:
		target = best
		target_changed.emit(target)

class_name OreCart
extends StaticBody2D
## A cart on a rail spur. Pushed from its start it rolls to the junction and takes
## whichever branch the points are set to: branch A ends at a buffer (haul it back
## and try again), branch B ends in the washout, where the cart wedges and becomes
## a bridge. State lives in GameState so it survives save/load.
## PROVISIONAL ART (drawn in code).

@export var state_key := "anjabi.cart"
@export var switch_key := "anjabi.spur_switch"
@export var rail_main: NodePath
@export var rail_a: NodePath
@export var rail_b: NodePath
## ProvBlock pit segment whose collision is removed once the cart bridges it.
@export var gap: NodePath
@export var bridge_position := Vector2.ZERO
@export var roll_speed := 70.0
@export var dialogue_buffer := ""
@export var dialogue_bridge := ""
@export var on_bridged: Array[Dictionary] = []

var moving := false

var _interactable: Interactable
var _shape: CollisionShape2D


func _ready() -> void:
	collision_layer = 1
	collision_mask = 0
	var rect := RectangleShape2D.new()
	rect.size = Vector2(18, 10)
	_shape = CollisionShape2D.new()
	_shape.shape = rect
	_shape.position = Vector2(0, -5)
	add_child(_shape)
	_interactable = Interactable.new()
	_interactable.label = "Ore cart"
	_interactable.radius = 16.0
	_interactable.position = Vector2(0, -5)
	add_child(_interactable)
	_interactable.interacted.connect(_on_interacted)
	_apply_state()


func state() -> String:
	return str(GameState.get_flag(state_key, "start"))


func _points(path: NodePath) -> PackedVector2Array:
	var rail := get_node(path) as ProvRail
	var out := PackedVector2Array()
	for p in rail.points:
		out.append(rail.to_global(p))
	return out


func _apply_state() -> void:
	var s := state()
	var bridged := s == "trench"
	match s:
		"buffer":
			var a := _points(rail_a)
			global_position = a[a.size() - 1]
		"trench":
			global_position = bridge_position
		_:
			global_position = _points(rail_main)[0]
	_shape.set_deferred("disabled", bridged)
	z_index = -4 if bridged else 0
	_interactable.enabled = not bridged
	_interactable.prompt = "Haul back" if s == "buffer" else "Push"
	var pit := get_node_or_null(gap) as ProvBlock
	if pit != null:
		pit.set_solid(not bridged)
	queue_redraw()


func _on_interacted(_by: Node2D) -> void:
	if moving:
		return
	var route := PackedVector2Array()
	var next := "start"
	var main := _points(rail_main)
	if state() == "start":
		var to_trench := str(GameState.get_flag(switch_key, "a")) == "b"
		route.append_array(main.slice(1))
		route.append_array(_points(rail_b if to_trench else rail_a).slice(1))
		if to_trench:
			route.append(bridge_position)
		next = "trench" if to_trench else "buffer"
	elif state() == "buffer":
		var a := _points(rail_a)
		a.reverse()
		route.append_array(a.slice(1))
		main.reverse()
		route.append_array(main.slice(1))
	else:
		return
	_roll(route, next)


func _roll(route: PackedVector2Array, next: String) -> void:
	moving = true
	_interactable.enabled = false
	var tween := create_tween()
	var from := global_position
	for i in route.size():
		var t := from.distance_to(route[i]) / roll_speed
		var step := tween.tween_property(self, "global_position", route[i], maxf(t, 0.05))
		if i == 0:
			step.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN)
		from = route[i]
	await tween.finished
	moving = false
	GameState.set_flag(state_key, next)
	_apply_state()
	if next == "trench":
		GameState.apply(on_bridged)
		if dialogue_bridge != "":
			Dialogue.start(dialogue_bridge)
	elif next == "buffer" and dialogue_buffer != "":
		Dialogue.start(dialogue_buffer)


func _draw() -> void:
	var rust := Color(0.5, 0.29, 0.18)
	if state() == "trench" and not moving:
		# Wedged in the washout, bed up: a short steel deck.
		draw_rect(Rect2(-13, -42, 26, 44), Color(0.23, 0.2, 0.19))
		draw_rect(Rect2(-12, -41, 24, 42), rust.darkened(0.2))
		for i in 6:
			draw_line(Vector2(-12, -38 + i * 7), Vector2(12, -38 + i * 7), rust.darkened(0.45), 1.0)
		return
	draw_rect(Rect2(-8, -3, 4, 3), Color(0.1, 0.1, 0.11))
	draw_rect(Rect2(4, -3, 4, 3), Color(0.1, 0.1, 0.11))
	draw_colored_polygon(PackedVector2Array([
		Vector2(-11, -15), Vector2(11, -15), Vector2(8, -3), Vector2(-8, -3)]), rust)
	draw_rect(Rect2(-11, -17, 22, 3), rust.lightened(0.12))
	draw_rect(Rect2(-9, -16, 18, 1), Color(0.08, 0.08, 0.09))
	draw_line(Vector2(-3, -13), Vector2(-4, -5), rust.darkened(0.3), 1.0)
	draw_line(Vector2(5, -13), Vector2(4, -6), rust.darkened(0.3), 1.0)

@export var dialogue_buffer := ""
@export var dialogue_bridge := ""

var rail_b := PackedVector3Array()

	var rust := StandardMaterial3D.new()

	rust.roughness = 0.65
	rust.metallic = 0.5
	var dark := StandardMaterial3D.new()
	dark.albedo_color = Color(0.1, 0.1, 0.11)
	dark.roughness = 0.5
	dark.metallic = 0.6
	_body = StaticBody3D.new()
	_body.collision_layer = 1
	_body.collision_mask = 0

	_mesh(_visual, Vector3(1.3, 0.08, 2.2), rust, Vector3(0, 1.0, 0))
	for x: float in [-0.5, 0.5]:

	_deck_body.add_child(_deck_visual)
	_mesh(_deck_visual, Vector3(1.3, 0.85, 2.4), rust, Vector3(0, -0.45, 0))

	_interactable.label = "Ore cart"

	_interactable.position = Vector3(0, 0.6, 0)

	for i in route.size():
		var t := from.distance_to(route[i]) / roll_speed
		var step := tween.tween_property(self, "global_position", route[i], maxf(t, 0.05))
		if i == 0:
			step.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN)
		from = route[i]
	if next == "trench":
		tween.tween_property(self, "global_position", bridge_position + Vector3(0, -0.9, 0), 0.25)
	await tween.finished
	moving = false
	GameState.set_flag(state_key, next)
	_apply_state()
	if next == "trench":
		GameState.apply(on_bridged)
		if dialogue_bridge != "":
			Dialogue.start(dialogue_bridge)
	elif next == "buffer" and dialogue_buffer != "":
		Dialogue.start(dialogue_buffer)


func _process(_delta: float) -> void:
	if moving:
		var d := Vector3.ZERO
		# face the direction of travel along the rails
		d = global_position - get_meta("last", global_position)
		set_meta("last", global_position)
		d.y = 0.0
		if d.length() > 0.001:
			_visual.rotation.y = atan2(d.x, d.z)


func _mesh(parent: Node3D, size: Vector3, m: Material, at: Vector3) -> void:
	var mi := MeshInstance3D.new()
	var b := BoxMesh.new()
	b.size = size
	mi.mesh = b
	mi.material_override = m
	mi.position = at
	parent.add_child(mi)

## Box spanning min..max corners.
func slab(a: Vector3, b: Vector3, m: Material, solid := true) -> Node3D:

	var hi := Vector3(maxf(a.x, b.x), maxf(a.y, b.y), maxf(a.z, b.z))
	return box((lo + hi) * 0.5, hi - lo, m, solid)

## Thin oriented box from a to b (walls, beams, rails). Height measured up from a.y.
func beam(a: Vector3, b: Vector3, h: float, t: float, m: Material, solid := false) -> Node3D:
	var d := b - a

	return n


## Stairs: bottom front-centre `start`, climbing along `dir` by `rise` over `run`.
## Visible solid steps plus one smooth invisible ramp collider for traversal.
func stairs(start: Vector3, dir: Vector3, rise: float, run: float, width: float, m: Material) -> void:
	dir = dir.normalized()
	var n := maxi(2, int(round(absf(rise) / 0.2)))
	var step_h := rise / n
	var step_d := run / n
	var yaw := atan2(dir.x, dir.z)
	for i in n:
		var top := start.y + step_h * (i + 1)
		var c := start + dir * step_d * (i + 0.5)
		var depth := top - (start.y - 0.4)
		box(Vector3(c.x, top - depth * 0.5, c.z), Vector3(width, depth, step_d + 0.01), m, false, yaw)
	# ramp collider whose surface runs through the step nosings
	var fwd := (dir * run + Vector3.UP * rise).normalized()
	var side := dir.cross(Vector3.UP).normalized()
	var up := side.cross(fwd).normalized()
	var length := Vector3(run, rise, 0).length() + 0.6
	var thick := 0.3
	var body := _body()
	# surface runs exactly from the bottom edge to the top edge; extra length only below
	body.transform = Transform3D(Basis(side, up, -fwd), start + dir * run * 0.5 + Vector3.UP * rise * 0.5 - fwd * 0.3 - up * thick * 0.5)
	var cs := CollisionShape3D.new()
	var s := BoxShape3D.new()
	s.size = Vector3(width, thick, length)
	cs.shape = s
	body.add_child(cs)
	add_child(body)


## Railing from a to b (feet level): posts, top rail, and an invisible 1.3 m wall.
func railing(a: Vector3, b: Vector3, m: Material = null) -> void:
	if m == null:
		m = M.steel_dark
	var len := a.distance_to(b)
	var posts := maxi(1, int(len / 1.6))
	for i in posts + 1:
		var p := a.lerp(b, float(i) / posts)
		box(p + Vector3(0, 0.5, 0), Vector3(0.08, 1.0, 0.08), m, false)
	beam(a + Vector3(0, 0.95, 0), b + Vector3(0, 0.95, 0), 0.06, 0.07, m)
	beam(a + Vector3(0, 0.5, 0), b + Vector3(0, 0.5, 0), 0.04, 0.05, m)
	beam(a, b, 1.3, 0.12, null, true)

class_name WalkMask
extends Node2D
## Walkable area stored as a BitMap, for maps laid out from a painted plan
## rather than from collision bodies. `pixel_size` is world pixels per mask pixel.

@export var mask: BitMap
@export var pixel_size := 2.0


func is_walkable(world: Vector2) -> bool:
	if mask == null:
		return true
	var p := Vector2i(((world - global_position) / pixel_size).floor())
	var s := mask.get_size()
	if p.x < 0 or p.y < 0 or p.x >= s.x or p.y >= s.y:
		return false
	return mask.get_bitv(p)


func nearest_walkable(world: Vector2, max_radius: float = 240.0) -> Vector2:
	if is_walkable(world):
		return world
	var r := pixel_size
	while r <= max_radius:
		for i in 16:
			var p := world + Vector2.from_angle(TAU * i / 16.0) * r
			if is_walkable(p):
				return p
		r += pixel_size
	return world

extends MapBuilder3D
## Anjabi / Collapsed Mine (location 04) in 3D. Replaces the 2D mine.tscn with the
## same layout at 20 px = 1 m: stairs down from the Estate in the north-west (6 m
## drop), a fenced inner yard under the cliff holding the sealed mine door, an outer
## yard with wash-house and sheds, the old spur line and its washout to the east,
## the path east toward the airpad, and the cliff drop to the south.
## Units: metres, +X east, -Z north (2D y -> Z), ground at y 0.

const TOP := 6.0
const TRENCH_FLOOR := -2.5
## Washout: x range and z range of the cut in the ground.
const TR_X0 := 37.2
const TR_X1 := 44.8
const TR_Z0 := 15.0
const TR_Z1 := 17.0


func _ready() -> void:
	_materials()
	_environment()
	_ground()
	_walls_and_fences()
	_mine_door()
	_buildings()
	_rails_and_puzzle()
	_props()
	_people()
	_interactables()


func _ground() -> void:
	# yard floor, with the washout cut out of it
	slab(Vector3(5.6, -1, 0), Vector3(45, 0, TR_Z0), M.gravel)
	slab(Vector3(5.6, -1, TR_Z1), Vector3(45, 0, 35.2), M.paving)
	slab(Vector3(5.6, -1, TR_Z0), Vector3(TR_X0, 0, TR_Z1), M.paving)
	slab(Vector3(TR_X1, -1, TR_Z0), Vector3(45, 0, TR_Z1), M.paving)
	# the washout itself
	slab(Vector3(TR_X0, TRENCH_FLOOR - 1, TR_Z0), Vector3(TR_X1, TRENCH_FLOOR, TR_Z1), M.rock_dark)
	slab(Vector3(TR_X0, TRENCH_FLOOR, TR_Z0 - 0.05), Vector3(TR_X1, 0, TR_Z0), M.rock, false)
	slab(Vector3(TR_X0, TRENCH_FLOOR, TR_Z1), Vector3(TR_X1, 0, TR_Z1 + 0.05), M.rock, false)
	var water := MeshInstance3D.new()
	var pm := PlaneMesh.new()
	pm.size = Vector2(TR_X1 - TR_X0, TR_Z1 - TR_Z0)
	water.mesh = pm
	water.material_override = M.sea
	water.position = Vector3((TR_X0 + TR_X1) * 0.5, TRENCH_FLOOR + 0.3, (TR_Z0 + TR_Z1) * 0.5)
	add_child(water)
	# mud paths: haul road, wash-house apron, path east, stair foot
	slab(Vector3(22.7, 0, 12.5), Vector3(25.3, 0.02, 34.5), M.mud, false)
	slab(Vector3(22.8, 0, 5.3), Vector3(25.2, 0.02, 12.5), M.mud, false)
	slab(Vector3(7.5, 0, 29.2), Vector3(22.6, 0.02, 31.2), M.mud, false)
	slab(Vector3(25.4, 0, 31.3), Vector3(45, 0.02, 33.1), M.mud, false)
	slab(Vector3(5.8, 0, 12.5), Vector3(22.6, 0.02, 14.6), M.mud, false)
	# the cliff drop beyond the south fence, and mist below
	slab(Vector3(0, -30, 35.2), Vector3(50, -0.5, 36.5), M.rock_dark)
	var mist := MeshInstance3D.new()
	var mp := PlaneMesh.new()
	mp.size = Vector2(300, 200)
	mist.mesh = mp
	mist.material_override = M.sea
	mist.position = Vector3(24, -24, 120)
	add_child(mist)

	var z := 5.3
	slab(Vector3(21.5, 0, z - 0.3), Vector3(26.5, 4.4, z), M.rock_dark, false)
	for x: float in [21.8, 26.2]:
		box(Vector3(x, 2.0, z + 0.2), Vector3(0.45, 4.0, 0.45), M.timber, true)
	box(Vector3(24, 4.2, z + 0.2), Vector3(5.2, 0.45, 0.5), M.timber, false)
	box(Vector3(24, 1.8, z + 0.05), Vector3(4.0, 3.6, 0.12), M.steel, true)
	box(Vector3(24, 1.8, z + 0.12), Vector3(0.05, 3.6, 0.04), M.steel_dark, false)
	# the bright weld bead round the frame
	var weld := _mat(Color(0.9, 0.85, 0.7), 0.3, 0.8, Color(0.6, 0.55, 0.4), 0.4)
	box(Vector3(24, 3.62, z + 0.13), Vector3(4.0, 0.05, 0.03), weld, false)
	box(Vector3(24, 0.05, z + 0.13), Vector3(4.0, 0.05, 0.03), weld, false)
	for x: float in [22.02, 25.98]:
		box(Vector3(x, 1.8, z + 0.13), Vector3(0.05, 3.6, 0.03), weld, false)
	sign_text(Vector3(24, 4.75, z + 0.5), "NO. 2 WORKING", 0.0, 44)
	# stacked rescue stretchers, still wrapped
	var wrap := _mat(Color(0.82, 0.8, 0.72), 0.7)
	for i in 3:
		box(Vector3(26.9, 0.12 + i * 0.16, 6.2), Vector3(0.7, 0.14, 2.0), wrap, i == 0, 0.08 * i)

func _rail_line(points: PackedVector3Array) -> void:
	for i in points.size() - 1:
		var a := points[i]
		var b := points[i + 1]
		var d := (b - a)
		var side := d.cross(Vector3.UP).normalized() * 0.35
		beam(a + side + Vector3(0, 0.02, 0), b + side + Vector3(0, 0.02, 0), 0.08, 0.07, M.steel)
		beam(a - side + Vector3(0, 0.02, 0), b - side + Vector3(0, 0.02, 0), 0.08, 0.07, M.steel)
		var n := int(a.distance_to(b) / 0.6)
		for j in n:
			var p := a.lerp(b, (j + 0.5) / n)
			box(p + Vector3(0, 0.02, 0), Vector3(1.1, 0.05, 0.22), M.timber, false, atan2(d.x, d.z))


func _rails_and_puzzle() -> void:
	var main := PackedVector3Array([Vector3(40, 0, 28.75), Vector3(40, 0, 22)])

== maps/anjabi/town_3d.gd

extends MapBuilder3D
## Anjabi, the whole settlement, in 3D. Terrain, cliffs, water and colliders come
## from maps/anjabi/town_3d_cells.res (built from Greybox Study 01 by
## tools/build_town_3d.gd): one cell per metre, walkway heights interpolated between
## the districts so the paths climb and fall between levels. Landmark buildings are
## placed on their greybox footprints. Announces the district the player is in.
## Greybox pixel p maps to world (p - ORIGIN) * PX metres.

const CELLS := "res://maps/anjabi/town_3d_cells.res"
const ORIGIN := Vector2(34, 34)
const PX := 0.12
const SEA := -7.0
const CHUNK := 16
const WATER := 0
const PATH := 1
const GRASS := 2
const SAND := 3
const SLAB := 4
const CLIFF := 5
const INSERT := 6
## Where the detailed Steps insert (district.tscn) sits; matches tools/build_town_3d.gd.
const INSERT_AT := Vector3(101.52, 0, 45.72)
const INSERT_SCENE := preload("res://maps/anjabi/district.tscn")

## Greybox-pixel rectangles; first match wins (same as the 2D town).
const DISTRICTS := [
	["THE STEPS", Rect2(820, 370, 125, 75)],
	["TRAM STATION", Rect2(940, 440, 180, 210)],
	["SHOPS", Rect2(890, 220, 170, 160)],
	["MAIN TAVERN", Rect2(670, 190, 215, 175)],
	["TION TOWER", Rect2(425, 300, 180, 175)],
	["GUARD TOWER", Rect2(930, 40, 210, 165)],
	["WORKER PODS", Rect2(440, 40, 325, 205)],
	["ESTATE GARDENS", Rect2(95, 320, 215, 100)],
	["ESTATE", Rect2(120, 420, 290, 165)],
	["COLLAPSED MINE", Rect2(60, 630, 250, 180)],
	["PRIVATE AIRPAD", Rect2(320, 735, 200, 130)],
	["BEACH", Rect2(500, 560, 430, 175)],

	["RAINFOREST APPROACH", Rect2(1255, 220, 165, 150)],
	["FOREST", Rect2(40, 40, 330, 215)],
]
## Spawn points and exits, in greybox pixels (the 2D town's positions / 2).
const SPAWNS := {"default": Vector2(880, 415), "mine": Vector2(212, 676), "airpad": Vector2(336, 746)}

var w := 0
var h := 0
var cls := PackedByteArray()
var hv := PackedFloat32Array()
var hc := PackedFloat32Array()
var _district := ""
var _acc := 0.0


func _ready() -> void:

	_materials()
	_environment()

	_collision()

# --- grid ------------------------------------------------------------------------

func _load_cells() -> void:
	var img: Image = load(CELLS)
	w = img.get_width()
	h = img.get_height()
	cls.resize(w * h)
	hv.resize(w * h)
	hc.resize(w * h)
	for z in h:
		for x in w:
			var c := img.get_pixel(x, z)
			var i := z * w + x
			hv[i] = c.r
			cls[i] = int(round(c.g))
			hc[i] = c.b


static func walkable(t: int) -> bool:
	return t == PATH or t == GRASS or t == SAND or t == INSERT


## World position of a greybox pixel, on the ground.
func at_px(p: Vector2) -> Vector3:
	var m := (p - ORIGIN) * PX
	return Vector3(m.x, ground(m.x, m.y), m.y)


func px_of(world: Vector3) -> Vector2:
	return Vector2(world.x, world.z) / PX + ORIGIN


## Visual height of the cell under (x, z).
func ground(x: float, z: float) -> float:
	var cx := clampi(int(x), 0, w - 1)
	var cz := clampi(int(z), 0, h - 1)
	return hv[cz * w + cx]


func nearest_walk(p: Vector2) -> Vector3:
	var m := (p - ORIGIN) * PX
	var c := Vector2i(int(m.x), int(m.y))
	for r in 12:
		for dz in range(-r, r + 1):
			for dx in range(-r, r + 1):
				if maxi(absi(dx), absi(dz)) != r:
					continue
				var q := c + Vector2i(dx, dz)
				if q.x < 0 or q.y < 0 or q.x >= w or q.y >= h:
					continue
				var i := q.y * w + q.x
				if walkable(cls[i]):
					return Vector3(q.x + 0.5, hv[i], q.y + 0.5)
	return Vector3(m.x, ground(m.x, m.y), m.y)


## Vertex heights: a vertex touching any walkable cell takes their mean, so walk
## surfaces stay exactly where the collider puts them; walls rise inside their cells.
func _vertex(x: int, z: int, collide: bool) -> float:
	var ws := 0.0
	var wn := 0
	var os := 0.0
	var on := 0
	var top := -INF
	for dz in [-1, 0]:
		for dx in [-1, 0]:
			var cx: int = x + dx
			var cz: int = z + dz
			if cx < 0 or cz < 0 or cx >= w or cz >= h:
				continue
			var i := cz * w + cx
			if cls[i] == INSERT:
				if collide:
					top = maxf(top, -30.0)
					continue
				os += hv[i] - 0.5
				on += 1
			elif walkable(cls[i]):
				ws += hv[i]
				wn += 1
			elif collide:
				top = maxf(top, hc[i])
			elif cls[i] != WATER:
				os += hv[i]
				on += 1
	if wn > 0:
		return ws / wn
	if collide:
		return top if top > -INF else 0.0
	return os / on if on > 0 else SEA


func _terrain() -> void:
	var vh := PackedFloat32Array()
	vh.resize((w + 1) * (h + 1))
	for z in h + 1:
		for x in w + 1:
			vh[z * (w + 1) + x] = _vertex(x, z, false)
	var mat := StandardMaterial3D.new()
	mat.vertex_color_use_as_albedo = true
	mat.vertex_color_is_srgb = true
	add_detail(mat, 0.3)
	mat.roughness = 0.72
	var rng := RandomNumberGenerator.new()
	rng.seed = 21
	for cz0 in range(0, h, CHUNK):
		for cx0 in range(0, w, CHUNK):
			var st := SurfaceTool.new()
			st.begin(Mesh.PRIMITIVE_TRIANGLES)
			st.set_smooth_group(-1)
			var any := false
			for z in range(cz0, mini(cz0 + CHUNK, h)):
				for x in range(cx0, mini(cx0 + CHUNK, w)):
					var i := z * w + x
					var t := cls[i]
					if t == WATER or t == INSERT:
						continue
					any = true
					var col := _colour(t, rng.randf())
					var a := Vector3(x, vh[z * (w + 1) + x], z)
					var b := Vector3(x + 1, vh[z * (w + 1) + x + 1], z)
					var c := Vector3(x + 1, vh[(z + 1) * (w + 1) + x + 1], z + 1)
					var d := Vector3(x, vh[(z + 1) * (w + 1) + x], z + 1)
					_quad(st, a, b, c, d, col)
					# skirts down to the sea where this cell borders water
					var dark := col.darkened(0.35)
					if z > 0 and cls[i - w] == WATER:
						_quad(st, Vector3(b.x, SEA, b.z), b, a, Vector3(a.x, SEA, a.z), dark)
					if z < h - 1 and cls[i + w] == WATER:
						_quad(st, Vector3(d.x, SEA, d.z), d, c, Vector3(c.x, SEA, c.z), dark)
					if x > 0 and cls[i - 1] == WATER:
						_quad(st, Vector3(a.x, SEA, a.z), a, d, Vector3(d.x, SEA, d.z), dark)
					if x < w - 1 and cls[i + 1] == WATER:
						_quad(st, Vector3(c.x, SEA, c.z), c, b, Vector3(b.x, SEA, b.z), dark)
			if not any:
				continue
			st.generate_normals()
			var mi := MeshInstance3D.new()
			mi.mesh = st.commit()
			mi.material_override = mat
			add_child(mi)


func _quad(st: SurfaceTool, a: Vector3, b: Vector3, c: Vector3, d: Vector3, col: Color) -> void:
	st.set_color(col)
	st.add_vertex(a)
	st.add_vertex(b)
	st.add_vertex(c)
	st.add_vertex(a)
	st.add_vertex(c)
	st.add_vertex(d)


func _colour(t: int, n: float) -> Color:
	var j := (n - 0.5) * 0.05
	match t:
		PATH:
			return Color(0.4 + j, 0.39 + j, 0.37 + j)
		GRASS:
			return Color(0.15 + j, 0.25 + j * 1.5, 0.18 + j)
		SAND:
			return Color(0.56 + j, 0.5 + j, 0.41 + j)
		SLAB:
			return Color(0.44 + j, 0.44 + j, 0.42 + j)
	return Color(0.21 + j, 0.24 + j, 0.26 + j)


func _collision() -> void:
	var data := PackedFloat32Array()

## Shed model filling a greybox footprint (visual only; the grid already blocks it).
func _shed_px(a: Vector2, b: Vector2, h_m: float) -> void:
	var lo := at_px(Vector2(minf(a.x, b.x), minf(a.y, b.y)))
	var hi := at_px(Vector2(maxf(a.x, b.x), maxf(a.y, b.y)))
	var base := Vector3((lo.x + hi.x) * 0.5, minf(lo.y, hi.y), (lo.z + hi.z) * 0.5)
	shed(base, Vector3(hi.x - lo.x, h_m, hi.z - lo.z), 0.0, false)


func _block(a: Vector2, b: Vector2, h_m: float, m: Material, roof: Material = null, lit := true) -> void:
	var lo := at_px(Vector2(minf(a.x, b.x), minf(a.y, b.y)))
	var hi := at_px(Vector2(maxf(a.x, b.x), maxf(a.y, b.y)))
	var base := minf(lo.y, hi.y)
	var size := Vector3(hi.x - lo.x, h_m, hi.z - lo.z)
	var c := Vector3((lo.x + hi.x) * 0.5, base + h_m * 0.5, (lo.z + hi.z) * 0.5)
	box(c, size, m, false)
	box(c + Vector3(0, h_m * 0.5 + 0.12, 0), size + Vector3(0.5, 0.24, 0.5), roof if roof != null else M.steel_dark, false)
	if lit:
		for i in maxi(1, int(size.x / 4.0)):
			box(Vector3(lo.x + 2.0 + i * 4.0, base + h_m * 0.55, hi.z + 0.04), Vector3(1.2, 0.8, 0.08), M.window, false)

func _vegetation_and_lamps() -> void:
	var rng := RandomNumberGenerator.new()
	rng.seed = 33
	var lamps := 0
	for z in range(1, h - 1):
		for x in range(1, w - 1):
			var i := z * w + x
			var t := cls[i]
			var r := rng.randf()
			if t == CLIFF and r < 0.03 and hv[i] > -2.0 and cls[i - 1] == CLIFF and cls[i + 1] == CLIFF \
					and cls[i - w] == CLIFF and cls[i + w] == CLIFF and cls[i - 2 * w] == CLIFF:
				tree(Vector3(x + 0.5, hv[i], z + 0.5), rng.randf_range(0.7, 1.1))
			elif t == GRASS and r < 0.012:
				var clear := true
				for n: int in [i - 1, i + 1, i - w, i + w, i - 2, i + 2, i - 2 * w, i + 2 * w]:
					if n < 0 or n >= w * h or cls[n] != GRASS:
						clear = false
				if clear:
					var p := Vector3(x + 0.5, hv[i], z + 0.5)
					tree(p, rng.randf_range(0.9, 1.4))
					blocker(p + Vector3(-0.25, 0, -0.25), p + Vector3(0.25, 1.6, 0.25))
			elif t == PATH and r < 0.03 and lamps < 48:
				# lamps stand at path edges, against a wall
				var edge := false
				for n: int in [i - 1, i + 1, i - w, i + w]:
					if cls[n] == CLIFF or cls[n] == SLAB:
						edge = true
				if edge:
					lamps += 1
					lamp(Vector3(x + 0.5, hv[i], z + 0.5), lamps % 2 == 0)

## Local z (relative to O) of the town walk cell at world x nearest the plaza centre line, or NAN.
func _walk_z(map: Node, x: float, z0: int, z1: int) -> float:
	var best := NAN
	for z in range(z0, z1 + 1):
		var cx := int(x)
		var cz := int(O.z + z)
		var t: int = map.cls[cz * map.w + cx]
		if t == 1 or t == 2 or t == 3:
			var local := (cz + 0.5) - O.z
			if is_nan(best) or absf(local) < absf(best):
				best = local
	return best


func _place3(p: Vector3) -> void:
	player.global_position = p
	player.velocity = Vector3.ZERO
	await _frames(6)


func check(ok: bool, what: String) -> void:
	_checks += 1
	if not ok:
		_fails += 1
	print("%s  %s" % ["PASS" if ok else "FAIL", what])


func _frames(n: int) -> void:
	for i in n:
		await get_tree().physics_frame
	await get_tree().process_frame


func _place(p: Vector2) -> void:
	player.global_position = p
	await _frames(4)


func _until(cond: Callable) -> void:
	for i in 1200:
		if cond.call():
			break
		await get_tree().physics_frame
	await _frames(2)


## Plays the open dialogue: picks the first choice containing any of `prefer`
## (each used once), otherwise the last choice, until it ends.
func _talk(prefer: Array) -> void:
	var want := prefer.duplicate()
	for i in 200:
		if not Dialogue.active:
			break
		var opts := Dialogue.choice_texts()
		if opts.is_empty():
			Dialogue.advance()
		else:
			var pick := opts.size() - 1
			for w: String in want:
				var found := false
				for j in opts.size():
					if opts[j].contains(w):
						pick = j
						want.erase(w)
						found = true
						break
				if found:
					break
			Dialogue.choose(pick)
		await get_tree().process_frame
	await _frames(2)

extends SceneTree
## Builds the cell grid for the 3D town from the director's greybox and its walk mask.
##   Godot --headless --path . -s tools/build_town_3d.gd
## One cell per metre (0.12 m per greybox pixel). Each cell gets a type (water, path,
## grass, sand, slab, cliff), a visual height and a collision height. Walkable heights
## are interpolated (Laplace) between district anchor heights, so walkways become
## gentle slopes between levels. Cliffs and slabs are raised walls in the collider.
## Writes maps/anjabi/town_3d_cells.res (Image, RGBF: r = visual h, g = type,
## b = collision h) and docs/reference/town_3d_heights_debug.png.
## Re-run after changing the greybox, the mask, or ANCHORS.

const SRC := "res://maps/anjabi/town_greybox.png"
const MASK := "res://maps/anjabi/town_walk.res"
const OUT := "res://maps/anjabi/town_3d_cells.res"
const DEBUG := "res://docs/reference/town_3d_heights_debug.png"
const ORIGIN := Vector2(34, 34)
const PX := 0.12
const W := 166
const H := 109
const SEA := -7.0
const WATER := 0
const PATH := 1
const GRASS := 2
const SAND := 3
const SLAB := 4
const CLIFF := 5
## Cells covered by the detailed Steps insert (maps/anjabi/district.gd placed at
## INSERT_AT): drawn and collided by the insert, not by the grid.
const INSERT := 6
## World position (m) of the insert's origin: the Steps, greybox (880, 415).
const INSERT_AT := Vector2(101.52, 45.72)
## Insert walk areas in insert-local metres (x0, z0, x1, z1) and their heights.
const INSERT_AREAS := [
	[Rect2(-12, -10, 32, 20), 0.0],   # the Steps plaza
	[Rect2(-24, -28, 36, 18), 3.0],   # tavern terrace
]
## Image px -> height (m). Walk cells within ANCHOR_R of each are fixed.
const ANCHORS := [
	[Vector2(880, 415), 0.0],    # the Steps
	[Vector2(930, 385), 0.0],    # shop front
	[Vector2(650, 250), 4.0],    # pods approach
	[Vector2(580, 70), 6.0],     # pods
	[Vector2(945, 130), 6.0],    # guard tower
	[Vector2(230, 110), 6.0],    # forest
	[Vector2(200, 345), 3.0],    # garden
	[Vector2(240, 578), 3.0],    # estate
	[Vector2(515, 465), 2.0],    # Tion Tower walk
	[Vector2(212, 700), -3.0],   # mine
	[Vector2(450, 800), -3.0],   # airpad
	[Vector2(650, 640), -6.0],   # beach
	[Vector2(1170, 460), -1.0],  # docks
	[Vector2(970, 465), -1.0],   # tram
	[Vector2(1380, 325), 0.0],   # rainforest approach
]
const ANCHOR_R := 4.0

var cls := PackedByteArray()
var hv := PackedFloat32Array()
var hc := PackedFloat32Array()
var insert_h := PackedFloat32Array()
var insert_fix := PackedByteArray()


func _init() -> void:
	var img := Image.load_from_file(ProjectSettings.globalize_path(SRC))
	img.convert(Image.FORMAT_RGB8)
	var mask: BitMap = load(MASK)
	cls.resize(W * H)
	hv.resize(W * H)
	hc.resize(W * H)
	var counts := [0, 0, 0, 0, 0, 0]
	for z in H:
		for x in W:
			var p := ORIGIN + (Vector2(x, z) + Vector2(0.5, 0.5)) / PX
			var votes := 0
			for o: Vector2 in [Vector2.ZERO, Vector2(2.5, 0), Vector2(-2.5, 0), Vector2(0, 2.5), Vector2(0, -2.5)]:
				var q := Vector2i(p + o)
				if mask.get_bitv(q):
					votes += 1
			var c := img.get_pixelv(Vector2i(p))
			var t := CLIFF
			if votes >= 3:
				t = GRASS if (c.g8 > c.r8 + 12 and c.g8 > c.b8 + 25) else PATH
			elif c.b8 > c.r8 + 30 and c.b8 > 170:
				t = WATER
			elif absi(c.r8 - c.g8) < 14 and absi(c.g8 - c.b8) < 16 and c.r8 > 168:
				t = SLAB
			cls[z * W + x] = t
			counts[t] += 1
	print("cells: water %d path %d grass %d sand %d slab %d cliff %d" % counts)
	# reserve the insert: cells inside it, and a one-cell rim pinned to its height
	insert_h.resize(W * H)
	insert_fix.resize(W * H)
	for a: Array in INSERT_AREAS:
		var r: Rect2 = a[0]
		r.position += INSERT_AT
		for z in range(int(r.position.y) - 1, int(r.end.y) + 2):
			for x in range(int(r.position.x) - 1, int(r.end.x) + 2):
				if x < 0 or z < 0 or x >= W or z >= H:
					continue
				var c := Vector2(x + 0.5, z + 0.5)
				var i := z * W + x
				if r.grow(-0.5).has_point(c):
					cls[i] = INSERT
					insert_h[i] = a[1]
					insert_fix[i] = 1
				elif r.grow(0.6).has_point(c) and cls[i] != INSERT and cls[i] != WATER and insert_fix[i] == 0:
					# a walkable rim so the insert always meets the town at its own height
					cls[i] = PATH
					insert_h[i] = a[1] - 0.03
					insert_fix[i] = 2
	_heights()
	_report()
	var out := Image.create(W, H, false, Image.FORMAT_RGBF)
	var dbg := Image.create(W, H, false, Image.FORMAT_RGB8)
	for z in H:
		for x in W:
			var i := z * W + x
			out.set_pixel(x, z, Color(hv[i], float(cls[i]), hc[i]))
			var k := clampf((hv[i] + 7.0) / 20.0, 0.0, 1.0)
			var col := Color(0.1, 0.25, 0.6) if cls[i] == WATER else Color(k, k, k)
			if _walk(cls[i]):
				col = Color(0.2 + k * 0.8, 0.6 * k + 0.2, 0.2)
			dbg.set_pixel(x, z, col)
	ResourceSaver.save(out, OUT)
	dbg.resize(W * 6, H * 6, Image.INTERPOLATE_NEAREST)
	dbg.save_png(ProjectSettings.globalize_path(DEBUG))
	print("wrote %s and %s" % [OUT, DEBUG])
	quit()


static func _walk(t: int) -> bool:
	return t == PATH or t == GRASS or t == SAND or t == INSERT


func _heights() -> void:
	var fixed := PackedByteArray()
	fixed.resize(W * H)
	for a: Array in ANCHORS:
		var c := (a[0] as Vector2 - ORIGIN) * PX
		for z in range(int(c.y - ANCHOR_R), int(c.y + ANCHOR_R) + 1):
			for x in range(int(c.x - ANCHOR_R), int(c.x + ANCHOR_R) + 1):
				if x < 0 or z < 0 or x >= W or z >= H:
					continue
				var i := z * W + x
				if _walk(cls[i]) and Vector2(x + 0.5, z + 0.5).distance_to(c) <= ANCHOR_R:
					fixed[i] = 1
					hv[i] = a[1]
	for i in W * H:
		if insert_fix[i] > 0:
			fixed[i] = 1
			hv[i] = insert_h[i]
	# successive over-relaxation over walk cells only
	for it in 4000:
		var worst := 0.0
		for z in H:
			for x in W:
				var i := z * W + x
				if fixed[i] == 1 or not _walk(cls[i]):
					continue
				var s := 0.0
				var n := 0
				if x > 0 and _walk(cls[i - 1]):
					s += hv[i - 1]; n += 1
				if x < W - 1 and _walk(cls[i + 1]):
					s += hv[i + 1]; n += 1
				if z > 0 and _walk(cls[i - W]):
					s += hv[i - W]; n += 1
				if z < H - 1 and _walk(cls[i + W]):
					s += hv[i + W]; n += 1
				if n == 0:
					continue
				var nv: float = hv[i] + 1.9 * (s / n - hv[i])
				worst = maxf(worst, absf(nv - hv[i]))
				hv[i] = nv
		if worst < 0.0005:
			print("relaxed in %d iterations" % it)
			break
	for i in W * H:
		if cls[i] == PATH and hv[i] < -4.6:
			cls[i] = SAND
	# non-walk cells: BFS from walk cells carrying the source height
	var base := PackedFloat32Array()
	base.resize(W * H)
	var dist := PackedInt32Array()
	dist.resize(W * H)
	dist.fill(-1)
	var q := PackedInt32Array()
	for i in W * H:
		if _walk(cls[i]):
			dist[i] = 0
			base[i] = hv[i]
			hc[i] = hv[i]
			q.append(i)
	var head := 0
	while head < q.size():
		var i := q[head]
		head += 1
		var x := i % W
		for n: int in [i - 1, i + 1, i - W, i + W]:
			if n < 0 or n >= W * H or dist[n] != -1:
				continue
			if (n == i - 1 and x == 0) or (n == i + 1 and x == W - 1):
				continue
			dist[n] = dist[i] + 1
			base[n] = base[i]
			q.append(n)
	for i in W * H:
		var t := cls[i]
		if t == INSERT:
			hc[i] = -30.0
			continue
		if _walk(t):
			continue
		var b := base[i] if dist[i] >= 0 else 0.0
		hc[i] = b + 4.0
		match t:
			WATER:
				hv[i] = SEA
			SLAB:
				hv[i] = b + 0.15
			_:
				hv[i] = b + minf(2.2 + dist[i] * 0.5, 5.0)


func _report() -> void:
	var worst := 0.0
	for z in H:
		for x in W - 1:
			var i := z * W + x
			if cls[i] == INSERT:
				continue
			if _walk(cls[i]) and _walk(cls[i + 1]) and cls[i + 1] != INSERT:
				worst = maxf(worst, absf(hv[i] - hv[i + 1]))
			if z < H - 1 and _walk(cls[i]) and _walk(cls[i + W]) and cls[i + W] != INSERT:
				worst = maxf(worst, absf(hv[i] - hv[i + W]))
	print("steepest walkable step between neighbouring cells: %.2f m (%.0f deg)" % [worst, rad_to_deg(atan(worst))])

extends SceneTree
## Builds the walkable mask for the Anjabi town greybox from the layout image itself.
##   Godot --headless --path . -s tools/build_town_mask.gd
## Walkways, beach, docks (beige) and vegetation (green) are walkable; cliffs,
## buildings and water are not. Light-grey slabs that are meant to be walked on
## cannot be told apart from buildings by colour, so they are listed in SLABS.
## Re-run after replacing maps/anjabi/town_greybox.png.

const SRC := "res://maps/anjabi/town_greybox.png"
const OUT := "res://maps/anjabi/town_walk.res"
const DEBUG := "res://docs/reference/town_walkable_debug.png"
const FRAME := Rect2i(34, 34, 1380, 902)
const GREEN_WALKABLE := true
const CLOSE_RADIUS := 3
## Image-pixel polygons forced walkable (mine yard, airpad).
const SLABS := [
	[Vector2(132, 655), Vector2(215, 642), Vector2(295, 668), Vector2(300, 715), Vector2(240, 768), Vector2(150, 755)],
	[Vector2(327, 790), Vector2(357, 750), Vector2(508, 750), Vector2(508, 826), Vector2(425, 850), Vector2(327, 800)],
]
## Landmarks (image pixels) checked for reachability from the first one.
const LANDMARKS := {
	"Steps": Vector2(880, 415), "Forest": Vector2(230, 110), "Garden": Vector2(200, 345),
	"Estate": Vector2(240, 578), "Mine": Vector2(212, 700), "Airpad": Vector2(450, 800),
	"Beach": Vector2(650, 640), "Tion Tower": Vector2(515, 465), "Tavern": Vector2(760, 350),
	"Pods": Vector2(580, 70), "Guard Tower": Vector2(945, 130), "Docks": Vector2(1170, 460),
	"Shop": Vector2(930, 385), "Tram": Vector2(970, 465), "Rainforest": Vector2(1380, 325),
}

var w := 0
var h := 0


func _init() -> void:
	var img := Image.load_from_file(ProjectSettings.globalize_path(SRC))
	img.convert(Image.FORMAT_RGB8)
	w = img.get_width()
	h = img.get_height()
	var px := img.get_data()
	var m := PackedByteArray()
	m.resize(w * h)
	for y in range(FRAME.position.y, FRAME.end.y):
		for x in range(FRAME.position.x, FRAME.end.x):
			var i := (y * w + x) * 3
			var r := px[i]
			var g := px[i + 1]
			var b := px[i + 2]
			var beige := r > 205 and g > 185 and b < 228 and r - b > 22 and r >= g
			var green := g > r + 12 and g > b + 25
			if beige or (GREEN_WALKABLE and green):
				m[y * w + x] = 1
	for poly: Array in SLABS:
		var pts := PackedVector2Array(poly)
		var box := Rect2(pts[0], Vector2.ZERO)
		for p in pts:
			box = box.expand(p)
		for y in range(int(box.position.y), int(box.end.y) + 1):
			for x in range(int(box.position.x), int(box.end.x) + 1):
				if Geometry2D.is_point_in_polygon(Vector2(x, y), pts):
					m[y * w + x] = 1
	# close small gaps: label text, stair treads, outlines
	m = _spread(m, CLOSE_RADIUS, 1)
	m = _spread(m, CLOSE_RADIUS, 0)
	for y in h:
		for x in w:
			if not FRAME.has_point(Vector2i(x, y)):
				m[y * w + x] = 0

	var bm := BitMap.new()
	bm.create(Vector2i(w, h))
	var dbg := Image.load_from_file(ProjectSettings.globalize_path(SRC))
	dbg.convert(Image.FORMAT_RGB8)
	var reach := _flood(m, _nearest(m, LANDMARKS["Steps"]))
	for y in h:
		for x in w:
			var i := y * w + x
			if m[i] == 1:
				bm.set_bit(x, y, true)
				dbg.set_pixel(x, y, Color(0.2, 0.9, 0.4) if reach[i] == 1 else Color(0.95, 0.2, 0.7))
			else:
				dbg.set_pixel(x, y, dbg.get_pixel(x, y).darkened(0.55))
	ResourceSaver.save(bm, OUT)
	dbg.save_png(ProjectSettings.globalize_path(DEBUG))
	for k: String in LANDMARKS:
		var p := _nearest(m, LANDMARKS[k])
		var ok := p.x >= 0 and reach[p.y * w + p.x] == 1
		print("%-12s %s  (snapped %.0f px)" % [k, "reachable" if ok else "NOT REACHABLE", Vector2(p).distance_to(LANDMARKS[k])])
	quit()


## Separable box dilation of cells equal to `value`.
func _spread(src: PackedByteArray, r: int, value: int) -> PackedByteArray:
	var tmp := src.duplicate()
	for y in h:
		var run := 0
		for x in range(-r, w):
			var add := x + r
			if add < w and src[y * w + add] == value:
				run += 1
			var drop := x - r - 1
			if drop >= 0 and src[y * w + drop] == value:
				run -= 1
			if x >= 0 and run > 0:
				tmp[y * w + x] = value
	var out := tmp.duplicate()
	for x in w:
		var run := 0
		for y in range(-r, h):
			var add := y + r
			if add < h and tmp[add * w + x] == value:
				run += 1
			var drop := y - r - 1
			if drop >= 0 and tmp[drop * w + x] == value:
				run -= 1
			if y >= 0 and run > 0:
				out[y * w + x] = value
	return out


func _nearest(m: PackedByteArray, at: Vector2) -> Vector2i:
	var c := Vector2i(at)
	for r in range(0, 60):
		for dy in range(-r, r + 1):
			for dx in range(-r, r + 1):
				if maxi(absi(dx), absi(dy)) != r:
					continue
				var p := c + Vector2i(dx, dy)
				if p.x >= 0 and p.y >= 0 and p.x < w and p.y < h and m[p.y * w + p.x] == 1:
					return p
	return Vector2i(-1, -1)


func _flood(m: PackedByteArray, start: Vector2i) -> PackedByteArray:
	var seen := PackedByteArray()
	seen.resize(w * h)
	if start.x < 0:
		return seen
	var stack := PackedInt32Array([start.y * w + start.x])
	seen[stack[0]] = 1
	while not stack.is_empty():
		var i := stack[stack.size() - 1]
		stack.remove_at(stack.size() - 1)
		var x := i % w
		for n: int in [i - 1, i + 1, i - w, i + w]:
			if n < 0 or n >= w * h or seen[n] == 1 or m[n] == 0:
				continue
			if (n == i - 1 and x == 0) or (n == i + 1 and x == w - 1):
				continue
			seen[n] = 1
			stack.append(n)
	return seen

#!/usr/bin/env python3
"""Concatenate every text source file into docs/ALL_SOURCE.txt for reviewers and
tools that cannot browse the repo. Run from the project root:
    python3 tools/bundle_source.py
"""
import os

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OUT = os.path.join(ROOT, "docs", "ALL_SOURCE.txt")
EXTS = (".gd", ".gdshader", ".json", ".tscn", ".godot", ".py", ".sh")
SKIP_DIRS = {".git", ".godot", "export", "docs"}

paths = []
for base, dirs, files in os.walk(ROOT):
    dirs[:] = sorted(d for d in dirs if d not in SKIP_DIRS)
    for f in sorted(files):
        if f.endswith(EXTS):
            paths.append(os.path.relpath(os.path.join(base, f), ROOT))

with open(OUT, "w") as out:
    out.write("COMETO - all source files. Index:\n")
    for p in paths:
        out.write("  %s\n" % p)
    for p in paths:
        out.write("\n\n" + "=" * 78 + "\n== %s\n" % p + "=" * 78 + "\n")
        with open(os.path.join(ROOT, p)) as f:
            out.write(f.read())
print("wrote %s (%d files)" % (os.path.relpath(OUT, ROOT), len(paths)))

	Dialogue.choices_shown.connect(_on_choices)


func _process(delta: float) -> void:
	if not _typing:
		return
	_shown += delta * chars_per_second
	if _shown >= _body.text.length():
		_finish_typing()
	else:
		_body.visible_characters = int(_shown)


func _input(event: InputEvent) -> void:
	if not visible or not event.is_action_pressed(&"interact"):
		return
	get_viewport().set_input_as_handled()
	if _choices.get_child_count() > 0:
		var f := get_viewport().gui_get_focus_owner() as Button
		if f != null and f.get_parent() == _choices:
			f.pressed.emit()
	else:
		_step()


func _gui_input(event: InputEvent) -> void:
	var mb := event as InputEventMouseButton
	if mb != null and mb.pressed and mb.button_index == MOUSE_BUTTON_LEFT:
		accept_event()
		if _choices.get_child_count() == 0:
			_step()


func _step() -> void:
	if _typing:
		_finish_typing()
	else:
		Dialogue.advance()


func _finish_typing() -> void:
	_typing = false
	_body.visible_characters = -1
	_more.visible = _choices.get_child_count() == 0


func _on_line(speaker: String, text: String) -> void:
	_clear_choices()
	_speaker.text = speaker.to_upper()
	_speaker.visible = speaker != ""
	_body.add_theme_color_override("default_color", UiTheme.BONE if speaker != "" else Color(0.72, 0.76, 0.78))
	_body.text = text
	_body.visible_characters = 0
	_shown = 0.0
	_typing = true
	_more.visible = false


func _on_choices(options: Array[String]) -> void:
	_finish_typing()
	_clear_choices()
	_more.visible = false
	for i in options.size():
		var b := Button.new()
		b.text = "›  " + options[i]
		b.alignment = HORIZONTAL_ALIGNMENT_LEFT
		b.pressed.connect(Dialogue.choose.bind(i))
		_choices.add_child(b)
	if not options.is_empty():
		(_choices.get_child(0) as Button).grab_focus.call_deferred()


func _on_ended(_id: String) -> void:
	_clear_choices()
	_typing = false
	visible = false


func _clear_choices() -> void:
	for c in _choices.get_children():
		_choices.remove_child(c)
		c.queue_free()

	_queue.append(text)
	if not _toasting:
		_next_toast()

	var t = player.get("target") if is_instance_valid(player) else null
	var show := t != null and not Dialogue.active
	_prompt.visible = show
	if show:
		var text: String = str(t.prompt).to_upper()

			text += "  ·  " + t.label
		if _prompt.text != text:
			_prompt.text = text

	if _queue.is_empty():
		_toasting = false
		return
	_toasting = true
	_toast.text = _queue.pop_front()
	var tw := create_tween()
	tw.tween_property(_toast, "modulate:a", 1.0, 0.25)
	tw.tween_interval(2.4)
	tw.tween_property(_toast, "modulate:a", 0.0, 0.5)
	tw.tween_callback(_next_toast)

		"objective_new":

		"objective":

			toast("CLUE RECORDED  ·  " + text)

			toast("JOURNAL UPDATED  ·  " + text)

## Keeps HUD chips clear of the notch and home indicator. Untested on device.
func _apply_safe_area() -> void:
	if not OS.has_feature("mobile"):
		return
	var win := Vector2(DisplayServer.window_get_size())
	if win.x <= 0.0 or win.y <= 0.0:
		return
	var safe := Rect2(DisplayServer.get_display_safe_area())
	var k := get_viewport_rect().size / win
	_safe.offset_left = safe.position.x * k.x
	_safe.offset_top = safe.position.y * k.y
	_safe.offset_right = -(win.x - safe.end.x) * k.x
	_safe.offset_bottom = -(win.y - safe.end.y) * k.y

	var dim := UiTheme.DIM.to_html(false)
	var q := ""
	for id: String in Journal.quests:
		var def: Dictionary = Journal.quest_defs.get(id, {})
		q += "[color=#%s]%s[/color]" % [amber, Journal.quest_title(id)]
		if Journal.is_complete(id):
			q += "  [color=#%s](to be continued)[/color]" % dim
		q += "\n[color=#%s]%s[/color]\n\n" % [dim, str(def.get("summary", ""))]
		for o: Dictionary in def.get("objectives", []):
			var st := Journal.objective_state(id, str(o["id"]))

	for id in Journal.clues:
		var def: Dictionary = Journal.clue_defs.get(id, {})

class_name TouchControls
extends Control
## Touch (and mouse) movement without on-screen buttons: drag anywhere to steer,
## tap to interact. Sits underneath every other control, so HUD chips and panels
## take their own touches first.

signal tapped

@export var deadzone := 5.0
@export var stick_radius := 22.0
@export var tap_ms := 350

var vector := Vector2.ZERO

var _down := false
var _dragged := false
var _origin := Vector2.ZERO
var _knob := Vector2.ZERO
var _t0 := 0


func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_STOP
	Dialogue.started.connect(func(_id: String) -> void: release())


func release() -> void:
	_down = false
	_dragged = false
	vector = Vector2.ZERO
	queue_redraw()


func _notification(what: int) -> void:
	if what == NOTIFICATION_PAUSED or what == NOTIFICATION_VISIBILITY_CHANGED:
		release()


func _gui_input(event: InputEvent) -> void:
	var mb := event as InputEventMouseButton
	if mb != null and mb.button_index == MOUSE_BUTTON_LEFT:
		if mb.pressed:
			_down = true
			_dragged = false
			_origin = mb.position
			_knob = mb.position
			_t0 = Time.get_ticks_msec()
		else:
			var was_tap := _down and not _dragged and Time.get_ticks_msec() - _t0 <= tap_ms
			release()
			if was_tap:
				tapped.emit()
		return
	var mm := event as InputEventMouseMotion
	if mm != null and _down:
		var d := mm.position - _origin
		if d.length() > deadzone:
			_dragged = true
		if _dragged:
			# floating stick: the origin trails the finger once it leaves the ring
			if d.length() > stick_radius:
				_origin = mm.position - d.normalized() * stick_radius
				d = mm.position - _origin
			vector = d / stick_radius
			_knob = mm.position
			queue_redraw()


func _draw() -> void:
	if not _dragged:
		return
	draw_arc(_origin, stick_radius, 0.0, TAU, 40, Color(1, 1, 1, 0.14), 1.0)
	draw_circle(_knob, 4.0, Color(1, 1, 1, 0.22))

static func box(bg: Color, border: Color, border_w: int = 1, pad: float = 6.0) -> StyleBoxFlat:
	var s := StyleBoxFlat.new()
	s.bg_color = bg
	s.border_color = border
	s.set_border_width_all(border_w)
	s.set_content_margin_all(pad)

static func build() -> Theme:
	var t := Theme.new()
	var f := SystemFont.new()
	f.font_names = PackedStringArray(["Avenir Next Condensed", "Avenir Next", "Helvetica Neue", "sans-serif"])
	f.font_weight = 500

	t.default_font_size = 10

	t.set_stylebox("panel", "PanelContainer", box(INK, Color(AMBER, 0.45)))

	var flat := box(Color(0, 0, 0, 0), Color(0, 0, 0, 0), 0, 3.0)
	var hover := box(Color(AMBER, 0.14), Color(0, 0, 0, 0), 0, 3.0)
	var focus := box(Color(AMBER, 0.1), Color(AMBER, 0.8), 0, 3.0)
	focus.border_width_left = 2
	t.set_stylebox("normal", "Button", flat)
	t.set_stylebox("hover", "Button", hover)
	t.set_stylebox("pressed", "Button", hover)
	t.set_stylebox("focus", "Button", focus)
	t.set_stylebox("disabled", "Button", flat)
	t.set_color("font_color", "Button", BONE)
	t.set_color("font_hover_color", "Button", AMBER)
	t.set_color("font_pressed_color", "Button", AMBER)
	t.set_color("font_focus_color", "Button", AMBER)
	t.set_color("font_disabled_color", "Button", Color(DIM, 0.5))

	t.set_constant("line_separation", "RichTextLabel", 1)
	t.set_constant("separation", "VBoxContainer", 2)
	t.set_constant("separation", "HBoxContainer", 6)

## A small bordered button for HUD chips.
